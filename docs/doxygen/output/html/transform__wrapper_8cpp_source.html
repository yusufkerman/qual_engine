110 glm::vec3 transform_wrapper::forward() const
111 {
112 const glm::vec3 rot = get_rotation_euler();
113 glm::mat4 rotMatrix(1.0f);
114 rotMatrix = glm::rotate(rotMatrix, rot.x, glm::vec3(1.0f, 0.0f, 0.0f));
115 rotMatrix = glm::rotate(rotMatrix, rot.y, glm::vec3(0.0f, 1.0f, 0.0f));
116 rotMatrix = glm::rotate(rotMatrix, rot.z, glm::vec3(0.0f, 0.0f, 1.0f));
117
118 // OpenGL sağ el kuralı: -Z ekseni ileri yöndür
119 return glm::normalize(-glm::vec3(rotMatrix[2]));
120 }

146 void transform_wrapper::look_at(const glm::vec3& target_point, const glm::vec3& world_up)
147 {
148 auto* t = raw();
149 if (!t)
150 {
151 return;
152 }
153
154 const glm::vec3 dir = target_point - t->position;
155 if (glm::length(dir) < 0.0001f)
156 {
157 return;
158 }
159
160 const glm::vec3 forward_dir = glm::normalize(dir);
161 const float pitch = -std::asin(std::clamp(forward_dir.y, -1.0f, 1.0f));
162 const float yaw = std::atan2(forward_dir.x, -forward_dir.z);
163
164 t->rotation = glm::vec3(pitch, yaw, 0.0f);
165 }
