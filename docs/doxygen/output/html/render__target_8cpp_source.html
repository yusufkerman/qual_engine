22void RenderTarget::create(int w, int h)
23{
24 width = clampSize(w);
25 height = clampSize(h);
26
27 if (fbo == 0) {
28 glGenFramebuffers(1, &fbo);
29 }
30
31 if (tex == 0) {
32 glGenTextures(1, &tex);
33 }
34
35 if (rbo == 0) {
36 glGenRenderbuffers(1, &rbo);
37 }
38
39 glBindFramebuffer(GL_FRAMEBUFFER, fbo);
40
41 // Renk eki olarak kullanılacak 2B dokuyu yapılandır
42 glBindTexture(GL_TEXTURE_2D, tex);
43 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
44 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
45 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
46 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
47 glTexImage2D(
48 GL_TEXTURE_2D,
49 0,
50 GL_RGB,
51 width,
52 height,
53 0,
54 GL_RGB,
55 GL_UNSIGNED_BYTE,
56 nullptr
57 );
58 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, tex, 0);
59
60 // Derinlik ve şablon (Depth/Stencil) tamponunu yapılandır
61 glBindRenderbuffer(GL_RENDERBUFFER, rbo);
62 glRenderbufferStorage(GL_RENDERBUFFER, GL_DEPTH24_STENCIL8, width, height);
63 glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_STENCIL_ATTACHMENT, GL_RENDERBUFFER, rbo);
64
65 const GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
66 if (status != GL_FRAMEBUFFER_COMPLETE)
67 {
68 std::cerr << "[RenderTarget] Framebuffer tamamlanamadi (Hata: " << status << ")\n";
69 }
70
71 glBindFramebuffer(GL_FRAMEBUFFER, 0);
72}
