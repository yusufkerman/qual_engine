42 state.prevKeys[i] = state.keys[i];
43 }
44
45 // Tuş durumlarını sorgula
46 if (state.active)
47 {
48 for (int i = 0; i <= GLFW_KEY_LAST; ++i)
49 {
50 state.keys[i] = (glfwGetKey(window, i) == GLFW_PRESS);
51 }
52 }
53 else
54 {

64 for (const auto& [key, action] : bindings)

67 {
68 continue;
69 }
70
71 const int a = static_cast<int>(action);
72
73 state.prevActions[a] = state.actions[a];
74 state.actions[a] = state.keys[key];
75 }
76
77 // Fare koordinatlarını ve delta farkını hesapla
78 double x = 0.0;
79 double y = 0.0;
80 glfwGetCursorPos(window, &x, &y);
81
82 const auto currentPos = glm::vec2(static_cast<float>(x), static_cast<float>(y));
83
84 if (state.firstMouse)
85 {
86 state.mousePos = currentPos;
87 state.mouseDelta = glm::vec2(0.0f, 0.0f);
88 state.gameMouseDelta = glm::vec2(0.0f, 0.0f);
89 state.firstMouse = false;
90 }
91 else
92 {
93 state.mouseDelta = currentPos - state.mousePos;
94
95 if (state.gameViewportSize.x > 1.0f && state.gameViewportSize.y > 1.0f)
96 {
97 state.gameMouseDelta = state.mouseDelta;
98 }
99 else
100 {
101 state.gameMouseDelta = state.mouseDelta;
102 }
103
104 state.mousePos = currentPos;
105 }
106}
