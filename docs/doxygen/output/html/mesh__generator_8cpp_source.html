68 generated_mesh_data generate_sphere(float radius, unsigned int sector_count, unsigned int stack_count)
69 {
70 generated_mesh_data data;
71
72 float x = 0.0f;
73 float y = 0.0f;
74 float z = 0.0f;
75 float xy = 0.0f;
76
77 const float length_inv = 1.0f / radius;
78 const float sector_step = 2.0f * glm::pi<float>() / static_cast<float>(sector_count);
79 const float stack_step = glm::pi<float>() / static_cast<float>(stack_count);
80
81 for (unsigned int i = 0; i <= stack_count; ++i)
82 {
83 const float stack_angle = (glm::pi<float>() * 0.5f) - (static_cast<float>(i) * stack_step);
84 xy = radius * std::cos(stack_angle);
85 z = radius * std::sin(stack_angle);
86
87 for (unsigned int j = 0; j <= sector_count; ++j)
88 {
89 const float sector_angle = static_cast<float>(j) * sector_step;
90
91 x = xy * std::cos(sector_angle);
92 y = xy * std::sin(sector_angle);
93
94 const float nx = x * length_inv;
95 const float ny = y * length_inv;
96 const float nz = z * length_inv;
97
98 const float s = static_cast<float>(j) / static_cast<float>(sector_count);
99 const float t = static_cast<float>(i) / static_cast<float>(stack_count);
100
101 data.vertices.push_back({ {x, y, z}, {nx, ny, nz}, {s, t} });
102 }
103 }
104
105 // İndis üretimi (Üçgen yüzey tanımlamaları)
106 for (unsigned int i = 0; i < stack_count; ++i)
107 {
108 unsigned int k1 = i * (sector_count + 1);
109 unsigned int k2 = k1 + sector_count + 1;
110
111 for (unsigned int j = 0; j < sector_count; ++j, ++k1, ++k2)
112 {
113 if (i != 0)
114 {
115 data.indices.push_back(k1);
116 data.indices.push_back(k2);
117 data.indices.push_back(k1 + 1);
118 }
119
120 if (i != (stack_count - 1))
121 {
122 data.indices.push_back(k1 + 1);
123 data.indices.push_back(k2);
124 data.indices.push_back(k2 + 1);
125 }
126 }
127 }
128
129 return data;
130 }
