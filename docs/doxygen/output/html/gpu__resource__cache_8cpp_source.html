32 gpu_texture_id gpu_resource_cache::get_texture_id(handle<texture_asset> h) {
33 if (auto it = m_texture_cache.find(h); it != m_texture_cache.end()) {
34 return it->second;
35 }
36
37 texture_asset* asset = m_registry->get_texture(h);
38 if (!asset || asset->raw_data.pixels.empty()) {
39 return m_fallback_texture_id;
40 }
41
42 gpu_texture_id id = m_backend->create_texture(asset->raw_data);
43 m_texture_cache[h] = id;
44
45 // Veri GPU'ya aktarıldığı için CPU tarafındaki ham diziyi (RAM) serbest bırak.
46 asset->raw_data.clear_raw_data();
47 return id;
48 }

50 submesh_runtime_info gpu_resource_cache::get_submesh_info(handle<model_asset> h, size_t sub_mesh_index) {
51 // 1. Önce GPU önbellek (cache) kontrolü
52 if (auto it = m_model_cache.find(h); it != m_model_cache.end()) {
53 if (sub_mesh_index < it->second.size()) {
54 return it->second[sub_mesh_index];
55 }
56 return submesh_runtime_info{};
57 }
58
59 // 2. İşlem için CPU varlığını talep et
60 model_asset* asset = m_registry->get_model(h);
61
62 if (!asset || asset->raw_data.sub_meshes.empty()) {
63 return submesh_runtime_info{};
64 }
65
66 // 3. Arka uç (Backend), CPU verilerini işleyip GPU nesnelerini oluşturur
67 std::vector<submesh_bridge> bridges = m_backend->create_model(asset->raw_data);
68
69 std::vector<submesh_runtime_info> runtime_infos;
70 runtime_infos.reserve(bridges.size());
71
72 for (size_t i = 0; i < bridges.size(); ++i) {
73 submesh_runtime_info info;
74
75 // GPU tanıtıcılarını ve sınır (bounds) verilerini aktar
76 info.gpu_id = bridges[i].gpu_data;
77 info.local_center = bridges[i].bounds.local_center;
78 info.local_radius = bridges[i].bounds.local_radius;
79
80 runtime_infos.push_back(info);
81
82 // Geriye dönük uyumluluk: Eski mesh yapısına GPU id'lerini senkronize et
83 if (i < asset->sub_meshes.size()) {
84 asset->sub_meshes[i].vao = bridges[i].gpu_data.vao_or_buffer_id;
85 asset->sub_meshes[i].vbo = bridges[i].gpu_data.vbo;
86 asset->sub_meshes[i].ebo = bridges[i].gpu_data.ebo;
87 }
88 }
89
90 m_model_cache[h] = runtime_infos;
91
92 // Model geometrisi GPU'ya aktarıldığı için CPU belleğini (RAM) temizle
93 asset->raw_data.clear_raw_data();
94
95 if (sub_mesh_index < runtime_infos.size()) {
96 return runtime_infos[sub_mesh_index];
97 }
98
99 return submesh_runtime_info{};
100 }

102 gpu_shader_id gpu_resource_cache::get_shader_id(handle<shader_asset> h) {
103 if (auto it = m_shader_cache.find(h); it != m_shader_cache.end()) {
104 return it->second;
105 }
106
107 if (!h.is_valid()) {
108 return m_fallback_shader_id;
109 }
110
111 shader_asset* asset = m_registry->get_shader(h);
112 if (!asset || asset->raw_data.vertex_source.empty()) {
113 return m_fallback_shader_id;
114 }
115
116 // Ham kaynak kodlarını (kurye verisini) donanım sürücüsüne teslim et
117 gpu_shader_id id = m_backend->create_shader(asset->raw_data);
118
119 // Derleme başarısız olursa 0 döner. Önbelleğe kaydetmeden yedek shader'ı kullan.
120 if (id == 0) {
121 return m_fallback_shader_id;
122 }
123
124 m_shader_cache[h] = id;
125
126 // Shader derlendiği ve donanıma geçtiği için RAM'deki metin dizilerini boşalt.
127 asset->raw_data.clear_raw_data();
128 return id;
129 }

159 void gpu_resource_cache::shutdown() {
160 // 1. Önbellekteki standart varlıkları temizle
161 clear_gpu_cache();
162
163 // 2. Yedek (Fallback) donanım kaynaklarını temizle
164 if (m_fallback_texture_id != 0) {
165 m_backend->destroy_texture(m_fallback_texture_id);
166 m_fallback_texture_id = 0;
167 }
168
169 if (m_fallback_shader_id != 0) {
170 m_backend->destroy_shader(m_fallback_shader_id);
171 m_fallback_shader_id = 0;
172 }
173
174 // 3. Arka uç sistemini kapat
175 m_backend->shutdown();
176 }

178 void gpu_resource_cache::clear_gpu_cache() {
179 // Dokuları temizle
180 for (auto const& [handle, id] : m_texture_cache) {
181 m_backend->destroy_texture(id);
182 }
183 m_texture_cache.clear();
184
185 // Modelleri temizle (Wrapper'dan çiğ GPU kimlikleri çıkarılarak arka uca iletilir)
186 for (auto const& [handle, runtime_infos] : m_model_cache) {
187 std::vector<gpu_mesh_id> ids_to_destroy;
188 ids_to_destroy.reserve(runtime_infos.size());
189
190 for (const auto& info : runtime_infos) {
191 ids_to_destroy.push_back(info.gpu_id);
192 }
193 m_backend->destroy_model(ids_to_destroy);
194 }
195 m_model_cache.clear();
196
197 // Shader'ları temizle
198 for (auto const& [handle, id] : m_shader_cache) {
199 m_backend->destroy_shader(id);
200 }
201 m_shader_cache.clear();
202 }
