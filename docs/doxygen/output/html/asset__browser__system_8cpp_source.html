35 asset_type asset_browser_system::get_type_from_extension(const std::filesystem::path& path) {
36 if (std::filesystem::is_directory(path)) return asset_type::folder;
37
38 std::string ext = path.extension().string();
39 std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
40
41 if (ext == ".png" || ext == ".jpg" || ext == ".tga" || ext == ".dds") return asset_type::texture;
42 if (ext == ".obj" || ext == ".fbx" || ext == ".gltf" || ext == ".dae") return asset_type::model;
43 if (ext == ".json" || ext == ".amat") return asset_type::material;
44
45 return asset_type::unknown;
46 }

48 void asset_browser_system::scan_directory(const std::filesystem::path& dir_path, asset_file_entry& out_entry)
49 {
50 for (const auto& entry : std::filesystem::directory_iterator(dir_path))
51 {
52 // .meta dosyaları arayüzde gösterilmez, doğrudan ait oldukları dosya ile okunur.
53 if (entry.path().extension() == ".meta")
54 continue;
55
56 asset_file_entry file_item;
57 file_item.full_path = entry.path();
58 file_item.relative_path = std::filesystem::relative(entry.path(), m_root_path);
59 file_item.name = entry.path().filename().string();
60 file_item.type = get_type_from_extension(entry.path());
61
62 // Varlık (Asset) boru hattından GUID sorgulanır (Yoksa yeni .meta oluşturulur)
63 file_item.guid = asset_pipeline::get_or_create_meta_guid(entry.path().string());
64
65 // Klasör değilse ve geçerli bir GUID'si varsa motorun adres defterine (registry) kaydet
66 if (file_item.type != asset_type::folder && m_aria_system && !file_item.guid.is_nil()) {
67 m_aria_system->register_asset_path(file_item.guid, entry.path().string());
68 }
69
70 if (file_item.type == asset_type::folder) {
71 scan_directory(entry.path(), file_item);
72 }
73
74 out_entry.children.push_back(file_item);
75 }
76
77 // Sıralama Mantığı: Klasörler her zaman üstte (A-Z), dosyalar altta (A-Z)
78 std::sort(out_entry.children.begin(), out_entry.children.end(),
79 [](const asset_file_entry& a, const asset_file_entry& b)
80 {
81 if (a.type == asset_type::folder && b.type != asset_type::folder) return true;
82 if (a.type != asset_type::folder && b.type == asset_type::folder) return false;
83 return a.name < b.name;
84 });
85 }

115 bool asset_browser_system::duplicate_asset(const std::filesystem::path& path) {
116 asset_type type = get_type_from_extension(path);
117
118 // Materyal işlemleri, içindeki GUID referanslarının da kopyalanması gerektiği için özel servise yönlendirilir.
119 if (type == asset_type::material && m_material_service) {
120 aria::asset_guid source_guid = asset_pipeline::get_or_create_meta_guid(path.string());
121 if (source_guid.is_nil()) return false;
122
123 aria::asset_guid new_guid = m_material_service->duplicate_material(source_guid, path.parent_path());
124 return !new_guid.is_nil();
125 }
126
127 // Standart dosya (doku, model vb.) kopyalama mantığı
128 try {
129 std::filesystem::path ext = path.extension();
130 std::filesystem::path dest = path.parent_path() / (path.stem().string() + "_Copy" + ext.string());
131
132 int count = 1;
133 while (std::filesystem::exists(dest)) {
134 dest = path.parent_path() / (path.stem().string() + "_Copy_" + std::to_string(count++) + ext.string());
135 }
136
137 if (std::filesystem::is_directory(path)) {
138 std::filesystem::copy(path, dest, std::filesystem::copy_options::recursive);
139 }
140 else {
141 std::filesystem::copy_file(path, dest);
142 }
143
144 return true;
145 }
146 catch (const std::exception& e) {
147 std::cerr << "[asset_browser_system] Dosya cogaltma hatasi (" << path << "): " << e.what() << "\n";
148 return false;
149 }
150 }
