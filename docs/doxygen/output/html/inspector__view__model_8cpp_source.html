108 std::vector<inspector_component_data> inspector_view_model::get_components() const
109 {
110 std::vector<inspector_component_data> out;
111
112 if (!has_selected_entity())
113 return out;
114
115 entity e = get_selected_entity();
116
117 // UI Registry'deki tüm kayıtlı arayüzleri tara
118 for (const auto& [hash, info] : m_world.get_component_registry())
119 {
120 // Varlık gerçekten bu bileşene sahipse UI verisini hazırla
121 if (m_world.has_component_by_hash(e, hash))
122 {
123 inspector_component_data data;
124 data.hash = hash;
125 data.ui = &info;
126 data.display_name = info.display_name;
127 data.default_open = true;
128 data.expanded = false;
129 data.removable = can_remove(hash);
130
131 out.push_back(std::move(data));
132 }
133 }
134
135 return out;
136 }

146 bool inspector_view_model::can_remove(size_t hash) const
147 {
148 if (!has_selected_entity())
149 return false;
150
151 if (!m_world.has_component_by_hash(get_selected_entity(), hash))
152 {
153 return false;
154 }
155
156 const ComponentUIInfo* info = m_world.get_component_ui(hash);
157 if (!info)
158 return false;
159
160 std::string name = info->display_name;
161
162 // TODO: (Mimari Not) İsim üzerinden kontrol etmek yerine, ileride bileşen kaydı sırasında
163 // "is_removable" gibi bir bayrak (flag) tanımlanması daha güvenli olacaktır.
164 if (name == "Transform" || name == "Name")
165 return false;
166
167 return true;
168 }
