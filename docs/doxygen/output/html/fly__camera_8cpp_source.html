14glm::vec3 calc_forward(const glm::vec3& rot)
15{
16 glm::vec3 forward;
17 // rot.x = Pitch (X ekseni etrafında dönüş)
18 // rot.y = Yaw (Y ekseni etrafında dönüş)
19 forward.x = std::cos(rot.x) * std::sin(rot.y);
20 forward.y = std::sin(rot.x);
21 forward.z = -std::cos(rot.x) * std::cos(rot.y); // OpenGL -Z derinlik standardı
22
23 return glm::normalize(forward);
24}

26void update_fly_camera(
27 GLFWwindow* window,
28 InputState& /*input*/, // TODO: Derleyici uyarısı almamak için isim gizlendi, ileride sisteme entegre edilebilir.
29 fly_camera_state& state,
30 glm::vec3& position,
31 glm::vec3& rotation,
32 float sensitivity,
33 float moveSpeed,
34 bool hoverGate,
35 float* scrollStep
36)
37{
38 if (!window)
39 return;
40
41 const bool rightHeld = (glfwGetMouseButton(window, GLFW_MOUSE_BUTTON_RIGHT) == GLFW_PRESS);
42 const bool shiftHeld = (glfwGetKey(window, GLFW_KEY_LEFT_SHIFT) == GLFW_PRESS);
43
44 // Shift tuşuna basılıysa hareketi hızlandır
45 const float speed = moveSpeed * (shiftHeld ? 3.0f : 1.0f);
46
47 // Sağ tıka İLK basıldığı an
48 if (rightHeld && !state.prevRightHeld && hoverGate)
49 {
50 state.capturingMouse = true;
51 state.firstMouse = true;
52 glfwSetInputMode(window, GLFW_CURSOR, GLFW_CURSOR_DISABLED);
53 }
54
55 // Sağ tık BIRAKILDIĞI an
56 if (!rightHeld && state.prevRightHeld)
57 {
58 state.capturingMouse = false;
59 state.firstMouse = true;
60 glfwSetInputMode(window, GLFW_CURSOR, GLFW_CURSOR_NORMAL);
61 }
62
63 state.prevRightHeld = rightHeld;
64
65 // Eğer fare kameraya kilitli değilse rotasyon/hareket hesaplama
66 if (!state.capturingMouse)
67 return;
68
69 double xpos, ypos;
70 glfwGetCursorPos(window, &xpos, &ypos);
71
72 if (state.firstMouse)
73 {
74 state.lastX = xpos;
75 state.lastY = ypos;
76 state.firstMouse = false;
77 }
78
79 const float xoffset = static_cast<float>(xpos - state.lastX);
80 const float yoffset = static_cast<float>(state.lastY - ypos);
81
82 state.lastX = xpos;
83 state.lastY = ypos;
84
85 // Fare hareketini rotasyona ekle
86 rotation.y += xoffset * sensitivity; // Yaw
87 rotation.x += yoffset * sensitivity; // Pitch
88
89 // Kameranın tepe taklak olmasını engelle (Gimbal lock koruması / Pitch sınırlaması)
90 rotation.x = std::clamp(rotation.x, -1.5f, 1.5f);
91
92 // Yön vektörlerini hesapla
93 const glm::vec3 forward = calc_forward(rotation);
94 const glm::vec3 right = glm::normalize(glm::cross(forward, glm::vec3(0.0f, 1.0f, 0.0f)));
95
96 // Klavye kontrolleri ile pozisyonu güncelle
97 if (glfwGetKey(window, GLFW_KEY_W) == GLFW_PRESS)
98 position += forward * speed;
99
100 if (glfwGetKey(window, GLFW_KEY_S) == GLFW_PRESS)
101 position -= forward * speed;
102
103 if (glfwGetKey(window, GLFW_KEY_D) == GLFW_PRESS)
104 position += right * speed;
105
106 if (glfwGetKey(window, GLFW_KEY_A) == GLFW_PRESS)
107 position -= right * speed;
108
109 // Fare tekerleği (scroll) kaydırıldıysa ileri/geri hızlı atılım yap
110 if (scrollStep)
111 {
112 position += forward * (*scrollStep) * 1.5f;
113 *scrollStep = 0.0f; // İşlem bittikten sonra sıfırla
114 }
115}
