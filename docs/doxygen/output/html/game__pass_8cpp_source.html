19void game_pass::render(EngineContext& ctx)
20{
21 int currentWidth = 0;
22 int currentHeight = 0;
23
24 // 1. Hedef Framebuffer ve Viewport sınırlarını belirle
25 if (ctx.isEditorBuild)
26 {
27 glBindFramebuffer(GL_FRAMEBUFFER, ctx.gameTarget.fbo);
28 glViewport(0, 0, ctx.gameTarget.width, ctx.gameTarget.height);
29
30 currentWidth = ctx.gameTarget.width;
31 currentHeight = ctx.gameTarget.height;
32 }
33 else
34 {
35 glBindFramebuffer(GL_FRAMEBUFFER, 0);
36 glViewport(0, 0, ctx.screenW, ctx.screenH);
37
38 currentWidth = ctx.screenW;
39 currentHeight = ctx.screenH;
40 }
41
42 glEnable(GL_DEPTH_TEST);
43 glDepthMask(GL_TRUE);
44 glClearColor(0.03f, 0.03f, 0.05f, 1.0f);
45 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
46
47 // 2. Sahnedeki aktif oyun kamerasını sorgula
48 scene& gameScene = ctx.session.game_scene();
49 const entity gameCam = gameScene.get_active_camera();
50
51 if (gameCam != 0 && gameScene.get_registry().has_component<camera_component>(gameCam))
52 {
53 const auto* t = gameScene.get_registry().get_component<transform_component>(gameCam);
54 const auto* c = gameScene.get_registry().get_component<camera_component>(gameCam);
55
56 if (t && c)
57 {
58 const glm::vec3 forward = calc_forward(t->rotation);
59
60 const glm::mat4 gameView = glm::lookAt(
61 t->position,
62 t->position + forward,
63 glm::vec3(0.0f, 1.0f, 0.0f)
64 );
65
66 // Dinamik en/boy oranı (sıfıra bölme koruması ile)
67 float aspectRatio = 16.0f / 9.0f;
68 if (currentHeight > 0)
69 {
70 aspectRatio = static_cast<float>(currentWidth) / static_cast<float>(currentHeight);
71 }
72
73 const glm::mat4 gameProj = glm::perspective(
74 glm::radians(c->fov),
75 aspectRatio,
76 c->nearPlane,
77 c->farPlane
78 );
79
80 // Gökyüzü çizimi
81 skybox_pass::render(ctx, gameView, gameProj);
82
83 // Ana sahne çizimi
84 ctx.renderSystem.render(
85 gameScene.get_registry(),
86 gameScene.get_light_batch(),
87 ctx,
88 gameView,
89 gameProj
90 );
91 }
92 }
93
94 if (ctx.isEditorBuild)
95 {
96 glBindFramebuffer(GL_FRAMEBUFFER, 0);
97 }
98}
