30 static raw_vertex process_assimp_vertex(aiMesh* mesh, unsigned int i) {
31 raw_vertex v;
32
33 v.position.x = mesh->mVertices[i].x;
34 v.position.y = mesh->mVertices[i].y;
35 v.position.z = mesh->mVertices[i].z;
36
37 if (mesh->HasNormals()) {
38 v.normal.x = mesh->mNormals[i].x;
39 v.normal.y = mesh->mNormals[i].y;
40 v.normal.z = mesh->mNormals[i].z;
41 }
42
43 if (mesh->mTextureCoords[0]) {
44 v.tex_coord.x = mesh->mTextureCoords[0][i].x;
45 v.tex_coord.y = mesh->mTextureCoords[0][i].y;
46 }
47
48 if (mesh->HasTangentsAndBitangents()) {
49 v.tangent.x = mesh->mTangents[i].x;
50 v.tangent.y = mesh->mTangents[i].y;
51 v.tangent.z = mesh->mTangents[i].z;
52 }
53
54 return v;
55 }

62 static void compute_mesh_bounds(raw_mesh_data& out_mesh) {
63 if (out_mesh.vertices.empty()) {
64 out_mesh.local_center = glm::vec3(0.0f);
65 out_mesh.local_radius = 0.0f;
66 return;
67 }
68
69 glm::vec3 min_b(std::numeric_limits<float>::max());
70 glm::vec3 max_b(std::numeric_limits<float>::lowest());
71
72 for (const auto& v : out_mesh.vertices) {
73 min_b = glm::min(min_b, v.position);
74 max_b = glm::max(max_b, v.position);
75 }
76
77 out_mesh.local_center = (min_b + max_b) * 0.5f;
78
79 float max_dist_sq = 0.0f;
80 for (const auto& v : out_mesh.vertices) {
81 glm::vec3 diff = v.position - out_mesh.local_center;
82 max_dist_sq = glm::max(max_dist_sq, glm::dot(diff, diff));
83 }
84
85 out_mesh.local_radius = glm::sqrt(max_dist_sq);
86 }

94 static raw_mesh_data process_assimp_mesh(aiMesh* mesh) {
95 raw_mesh_data out_mesh;
96
97 out_mesh.vertices.reserve(mesh->mNumVertices);
98 for (unsigned int i = 0; i < mesh->mNumVertices; i++) {
99 out_mesh.vertices.push_back(process_assimp_vertex(mesh, i));
100 }
101
102 for (unsigned int i = 0; i < mesh->mNumFaces; i++) {
103 aiFace face = mesh->mFaces[i];
104 for (unsigned int j = 0; j < face.mNumIndices; j++) {
105 out_mesh.indices.push_back(face.mIndices[j]);
106 }
107 }
108
109 compute_mesh_bounds(out_mesh);
110 return out_mesh;
111 }

120 static void process_assimp_node(aiNode* node, const aiScene* scene, raw_model_data& out_model) {
121 for (unsigned int i = 0; i < node->mNumMeshes; i++) {
122 aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
123 out_model.sub_meshes.push_back(process_assimp_mesh(mesh));
124 }
125
126 for (unsigned int i = 0; i < node->mNumChildren; i++) {
127 process_assimp_node(node->mChildren[i], scene, out_model);
128 }

131 std::optional<raw_texture_data> asset_loader::load_texture_to_cpu(const std::string& file_path) {
132 if (file_path.empty()) {
133 std::cout << "[ARIA Loader] Hata: Doku dosya yolu bos.\n";
134 return std::nullopt;
135 }
136
137 int width = 0, height = 0, channels = 0;
138 stbi_set_flip_vertically_on_load(true);
139 unsigned char* data = stbi_load(file_path.c_str(), &width, &height, &channels, 0);
140
141 if (!data) {
142 std::cout << "[ARIA Loader] Hata: Doku dosyasi yuklenemedi: " << file_path << "\n";
143 return std::nullopt;
144 }
145
146 raw_texture_data out_raw_data;
147 out_raw_data.debug_path = file_path;
148 out_raw_data.width = width;
149 out_raw_data.height = height;
150 out_raw_data.channels = channels;
151
152 size_t total_size = static_cast<size_t>(width) * height * channels;
153 out_raw_data.pixels.resize(total_size);
154 std::memcpy(out_raw_data.pixels.data(), data, total_size);
155
156 stbi_image_free(data);
157 return out_raw_data;
158 }

160 std::optional<raw_model_data> asset_loader::load_model_to_cpu(const std::string& file_path) {
161 if (file_path.empty()) {
162 std::cout << "[ARIA Loader] Hata: Model dosya yolu bos.\n";
163 return std::nullopt;
164 }
165
166 Assimp::Importer importer;
167 const aiScene* scene = importer.ReadFile(
168 file_path,
169 aiProcess_Triangulate |
170 aiProcess_GenNormals |
171 aiProcess_JoinIdenticalVertices |
172 aiProcess_CalcTangentSpace
173 );
174
175 if (!scene || !scene->mRootNode) {
176 std::cout << "[ARIA Loader] Hata: Assimp yukleme basarisiz oldu: " << file_path << "\n";
177 std::cout << "[Assimp Hata Detayi] " << importer.GetErrorString() << std::endl;
178 return std::nullopt;
179 }
180
181 raw_model_data out_model_data;
182 out_model_data.debug_path = file_path;
183
184 process_assimp_node(scene->mRootNode, scene, out_model_data);
185
186 return out_model_data;
187 }

189 std::optional<raw_material_data> asset_loader::load_material_to_cpu(const std::string& file_path) {
190 if (file_path.empty()) {
191 std::cout << "[ARIA Loader] Hata: Materyal dosya yolu bos.\n";
192 return std::nullopt;
193 }
194
195 std::ifstream file(file_path);
196 if (!file.is_open()) {
197 std::cout << "[ARIA Loader] Hata: Materyal dosyasi acilamadi: " << file_path << "\n";
198 return std::nullopt;
199 }
200
201 json material_json;
202 try {
203 file >> material_json;
204 }
205 catch (const json::parse_error& e) {
206 std::cout << "[ARIA Loader] Hata: JSON ayristirma basarisiz: " << e.what() << "\n";
207 file.close();
208 return std::nullopt;
209 }
210 file.close();
211
212 try {
213 raw_material_data out_raw_data;
214 out_raw_data.debug_path = file_path;
215 out_raw_data.name = material_json.value("name", "Unnamed_Material");
216
217 // Render Modu Değerlendirmesi
218 if (material_json.contains("render_mode")) {
219 int mode_int = material_json.value("render_mode", 0);
220 if (mode_int == 1) {
221 out_raw_data.mode = render_mode::cutout;
222 }
223 else if (mode_int == 2) {
224 out_raw_data.mode = render_mode::transparent;
225 }
226 else {
227 out_raw_data.mode = render_mode::opaque;
228 }
229 }
230 else {
231 out_raw_data.mode = render_mode::opaque;
232 }
233
234 out_raw_data.alpha_threshold = material_json.value("alpha_threshold", 0.1f);
235
236 // Temel Renk (Base Color) Değerlendirmesi
237 if (material_json.contains("base_color")) {
238 auto& bc = material_json["base_color"];
239 if (bc.is_array()) {
240 out_raw_data.base_color.x = bc.size() > 0 ? bc[0].get<float>() : 1.0f;
241 out_raw_data.base_color.y = bc.size() > 1 ? bc[1].get<float>() : 1.0f;
242 out_raw_data.base_color.z = bc.size() > 2 ? bc[2].get<float>() : 1.0f;
243 }
244 else if (bc.is_object()) {
245 out_raw_data.base_color.x = bc.value("x", 1.0f);
246 out_raw_data.base_color.y = bc.value("y", 1.0f);
247 out_raw_data.base_color.z = bc.value("z", 1.0f);
248 }
249 }
250 else {
251 out_raw_data.base_color = glm::vec3(1.0f, 1.0f, 1.0f);
252 }
253
254 // Standart Özellikler
255 out_raw_data.specular_strength = material_json.value("specular_strength", 0.5f);
256 out_raw_data.shininess = material_json.value("shininess", 32.0f);
257 out_raw_data.normal_strength = material_json.value("normal_strength", 1.0f);
258
259 // Doku Ölçeklendirmesi (Texture Scale) Değerlendirmesi
260 if (material_json.contains("texture_scale")) {
261 auto& ts = material_json["texture_scale"];
262 if (ts.is_object()) {
263 out_raw_data.texture_scale.x = ts.value("x", 1.0f);
264 out_raw_data.texture_scale.y = ts.value("y", 1.0f);
265 }
266 else if (ts.is_array()) {
267 out_raw_data.texture_scale.x = ts.size() > 0 ? ts[0].get<float>() : 1.0f;
268 out_raw_data.texture_scale.y = ts.size() > 1 ? ts[1].get<float>() : 1.0f;
269 }
270 }
271 else {
272 out_raw_data.texture_scale = glm::vec2(1.0f, 1.0f);
273 }
274
275 // JSON içindeki string formatlı GUID'leri stduuid yapısına güvenle çeviren yerel fonksiyon
276 auto parse_guid_helper = [](const json& j, const std::string& key) -> aria::asset_guid {
277 std::string guid_str = j.value(key, "");
278 if (guid_str.empty()) return {};
279
280 auto parsed = uuids::uuid::from_string(guid_str);
281 return parsed ? *parsed : aria::asset_guid{};
282 };
283
284 // Referans GUID Atamaları
285 out_raw_data.shader_guid = parse_guid_helper(material_json, "shader_guid");
286 out_raw_data.diffuse_texture_guid = parse_guid_helper(material_json, "diffuse_texture_guid");
287 out_raw_data.normal_texture_guid = parse_guid_helper(material_json, "normal_texture_guid");
288 out_raw_data.specular_texture_guid = parse_guid_helper(material_json, "specular_texture_guid");
289
290 return out_raw_data;
291 }
292 catch (const nlohmann::json::exception& e) {
293 std::cout << "[ARIA Loader] Hata: JSON Tip Uyumsuzlugu (Runtime): " << e.what() << "\n";
294 return std::nullopt;
295 }
296 }

298 std::optional<raw_shader_data> asset_loader::load_shader_to_cpu(const std::string& file_path) {
299 std::string vert_path = file_path + ".vert";
300 std::string frag_path = file_path + ".frag";
301
302 std::ifstream vFile(vert_path);
303 std::ifstream fFile(frag_path);
304
305 if (!vFile.is_open()) {
306 std::cout << "[ARIA Loader] Hata: Vertex shader dosyasi bulunamadi: " << vert_path << std::endl;
307 }
308 if (!fFile.is_open()) {
309 std::cout << "[ARIA Loader] Hata: Fragment shader dosyasi bulunamadi: " << frag_path << std::endl;
310 }
311
312 if (!vFile.is_open() || !fFile.is_open()) {
313 return std::nullopt;
314 }
315
316 raw_shader_data raw;
317 raw.debug_path = file_path;
318
319 std::stringstream vStr, fStr;
320 vStr << vFile.rdbuf();
321 fStr << fFile.rdbuf();
322
323 raw.vertex_source = vStr.str();
324 raw.fragment_source = fStr.str();
325
326 vFile.close();
327 fFile.close();
328
329 return raw;
330 }
