19 Vertex ProcessVertex(const aiMesh* mesh, unsigned int i)
20 {
21 Vertex v;
22
23 v.Position = glm::vec3(mesh->mVertices[i].x, mesh->mVertices[i].y, mesh->mVertices[i].z);
24
25 if (mesh->HasNormals())
26 {
27 v.Normal = glm::vec3(mesh->mNormals[i].x, mesh->mNormals[i].y, mesh->mNormals[i].z);
28 }
29 else
30 {
31 v.Normal = glm::vec3(0.0f);
32 }
33
34 if (mesh->mTextureCoords[0])
35 {
36 v.TexCoord = glm::vec2(mesh->mTextureCoords[0][i].x, mesh->mTextureCoords[0][i].y);
37 }
38 else
39 {
40 v.TexCoord = glm::vec2(0.0f);
41 }
42
43 if (mesh->HasTangentsAndBitangents())
44 {
45 v.Tangent = glm::vec3(mesh->mTangents[i].x, mesh->mTangents[i].y, mesh->mTangents[i].z);
46 }
47 else
48 {
49 v.Tangent = glm::vec3(0.0f);
50 }
51
52 return v;
53 }

58 void ComputeBounds(Mesh& out)
59 {
60 if (out.vertices.empty())
61 {
62 out.localCenter = glm::vec3(0.0f);
63 out.localRadius = 0.0f;
64 return;
65 }
66
67 glm::vec3 minB(std::numeric_limits<float>::max());
68 glm::vec3 maxB(std::numeric_limits<float>::lowest());
69
70 for (const auto& v : out.vertices)
71 {
72 minB = glm::min(minB, v.Position);
73 maxB = glm::max(maxB, v.Position);
74 }
75
76 out.localCenter = (minB + maxB) * 0.5f;
77
78 float maxDistSq = 0.0f;
79 for (const auto& v : out.vertices)
80 {
81 glm::vec3 diff = v.Position - out.localCenter;
82 maxDistSq = glm::max(maxDistSq, glm::dot(diff, diff));
83 }
84
85 out.localRadius = glm::sqrt(maxDistSq);
86 }

91 Mesh ProcessMesh(const aiMesh* mesh)
92 {
93 Mesh out;
94 out.name = mesh->mName.C_Str();
95
96 out.vertices.reserve(mesh->mNumVertices);
97 for (unsigned int i = 0; i < mesh->mNumVertices; ++i)
98 {
99 out.vertices.push_back(ProcessVertex(mesh, i));
100 }
101
102 out.indices.reserve(mesh->mNumFaces * 3);
103 for (unsigned int i = 0; i < mesh->mNumFaces; ++i)
104 {
105 const aiFace& face = mesh->mFaces[i];
106 for (unsigned int j = 0; j < face.mNumIndices; ++j)
107 {
108 out.indices.push_back(face.mIndices[j]);
109 }
110 }
111
112 ComputeBounds(out);
113 return out;
114 }

119 void ProcessNode(const aiNode* node, const aiScene* scene, Model& model)
120 {
121 for (unsigned int i = 0; i < node->mNumMeshes; ++i)
122 {
123 const aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
124 model.meshes.push_back(ProcessMesh(mesh));
125 }
126
127 for (unsigned int i = 0; i < node->mNumChildren; ++i)
128 {
129 ProcessNode(node->mChildren[i], scene, model);
130 }
131 }

135Model ModelLoader::Load(const std::string& path)
136{
137 Assimp::Importer importer;
138
139 const aiScene* scene = importer.ReadFile(
140 path,
141 aiProcess_Triangulate |
142 aiProcess_GenNormals |
143 aiProcess_JoinIdenticalVertices |
144 aiProcess_CalcTangentSpace
145 );
146
147 Model model;
148
149 if (!scene || (scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE) || !scene->mRootNode)
150 {
151 std::cerr << "[ModelLoader] Model yuklenemedi: " << path << '\n'
152 << "[Assimp Hatasi] " << importer.GetErrorString() << '\n';
153 return model;
154 }
155
156 ProcessNode(scene->mRootNode, scene, model);
157 return model;
158}
