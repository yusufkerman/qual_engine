20 static raw_mesh_data convert_generator_to_raw_mesh(const generated_mesh_data& generated) {
21 raw_mesh_data out;
22 out.vertices.reserve(generated.vertices.size());
23 out.indices = generated.indices;
24
25 glm::vec3 minB(std::numeric_limits<float>::max());
26 glm::vec3 maxB(std::numeric_limits<float>::lowest());
27
28 for (const auto& g_vert : generated.vertices) {
29 raw_vertex v;
30 v.position = g_vert.position;
31 v.normal = g_vert.normal;
32 v.tex_coord = g_vert.tex_coords;
33 v.tangent = glm::vec3(0.0f);
34
35 out.vertices.push_back(v);
36 minB = glm::min(minB, g_vert.position);
37 maxB = glm::max(maxB, g_vert.position);
38 }
39
40 out.local_center = (minB + maxB) * 0.5f;
41 float maxDistSq = 0.0f;
42 for (const auto& v : out.vertices) {
43 glm::vec3 diff = v.position - out.local_center;
44 maxDistSq = glm::max(maxDistSq, glm::dot(diff, diff));
45 }
46 out.local_radius = glm::sqrt(maxDistSq);
47 return out;
48 }

54 handle<model_asset> asset_registry::load_model(const asset_guid& guid, const std::string& file_path) {
55 // Önbellek (Cache) kontrolü
56 if (auto it = m_model_map.find(guid); it != m_model_map.end()) {
57 return it->second;
58 }
59
60 raw_model_data modelData;
61 modelData.debug_path = file_path;
62
63 // Sanal Yol (Virtual File System) Kontrolü
64 if (file_path == "engine://builtin/cube") {
65 generated_mesh_data raw_cube = mesh_generator::generate_cube();
66 modelData.sub_meshes.push_back(convert_generator_to_raw_mesh(raw_cube));
67 return add_procedural_model(guid, modelData);
68 }
69 else if (file_path.rfind("engine://builtin/sphere_", 0) == 0) {
70 // Yoldan yarıçap değerini güvenli bir şekilde ayrıştır
71 std::string radius_str = file_path.substr(24); // "engine://builtin/sphere_" uzunluğu
72 float radius = std::stof(radius_str);
73
74 generated_mesh_data raw_sphere = mesh_generator::generate_sphere(radius, 36, 18);
75 modelData.sub_meshes.push_back(convert_generator_to_raw_mesh(raw_sphere));
76 return add_procedural_model(guid, modelData);
77 }
78
79 // Gerçek Dosya Yükleme Süreci
80 auto loaded = asset_loader::load_model_to_cpu(file_path);
81 if (!loaded) return handle<model_asset>{};
82
83 return add_procedural_model(guid, *loaded);
84 }

86 handle<model_asset> asset_registry::add_procedural_model(const asset_guid& guid, const raw_model_data& raw_data) {
87 uint32_t slot_idx;
88 uint32_t gen = 1;
89
90 if (!m_model_free_slots.empty()) {
91 slot_idx = m_model_free_slots.back();
92 m_model_free_slots.pop_back();
93 gen = m_model_slots[slot_idx].generation;
94 }
95 else {
96 m_model_slots.emplace_back();
97 slot_idx = static_cast<uint32_t>(m_model_slots.size() - 1);
98 }
99
100 auto& slot = m_model_slots[slot_idx];
101 slot.is_allocated = true;
102 slot.asset.guid = guid;
103 slot.asset.raw_data = raw_data;
104
105 // Mesh meta veri paketini (sub_meshes dizisini) kuryeden ayıklayıp dolduruyoruz
106 slot.asset.sub_meshes.resize(raw_data.sub_meshes.size());
107 for (size_t i = 0; i < raw_data.sub_meshes.size(); ++i) {
108 slot.asset.sub_meshes[i].index_count = static_cast<unsigned int>(raw_data.sub_meshes[i].indices.size());
109 slot.asset.sub_meshes[i].local_center = raw_data.sub_meshes[i].local_center;
110 slot.asset.sub_meshes[i].local_radius = raw_data.sub_meshes[i].local_radius;
111 }
112
113 handle<model_asset> h{ slot_idx, gen };
114 m_model_map[guid] = h;
115 return h;
116 }

140 handle<texture_asset> asset_registry::load_texture(const asset_guid& guid) {
141 if (auto it = m_texture_map.find(guid); it != m_texture_map.end())
142 return it->second;
143
144 if (guid.is_nil()) return {};
145
146 if (!m_database) {
147 std::cout << "[ARIA Registry] Hata: Varlik veritabani (asset_database) tanimsiz!\n";
148 return {};
149 }
150
151 std::string path = m_database->resolve_path(guid);
152
153 if (path.empty()) {
154 std::cout << "[ARIA Registry] Uyari: Doku GUID'si veritabaninda bulunamadi.\n";
155 return {};
156 }
157
158 auto loaded = asset_loader::load_texture_to_cpu(path);
159 if (!loaded) return {};
160
161 uint32_t idx = m_texture_free_slots.empty()
162 ? (m_texture_slots.emplace_back(), m_texture_slots.size() - 1)
163 : (m_texture_free_slots.back());
164
165 if (!m_texture_free_slots.empty()) m_texture_free_slots.pop_back();
166
167 auto& slot = m_texture_slots[idx];
168 slot.is_allocated = true;
169 slot.asset.guid = guid;
170 slot.asset.raw_data = *loaded;
171 slot.asset.width = loaded->width;
172 slot.asset.height = loaded->height;
173 slot.asset.channels = loaded->channels;
174
175 return m_texture_map[guid] = handle<texture_asset>{ idx, slot.generation };
176 }

198 handle<material_asset> asset_registry::load_material(const asset_guid& guid, const std::string& file_path) {
199 if (auto it = m_material_map.find(guid); it != m_material_map.end())
200 return it->second;
201
202 auto loaded = asset_loader::load_material_to_cpu(file_path);
203 if (!loaded)
204 return {};
205
206 uint32_t idx;
207 if (m_material_free_slots.empty()) {
208 m_material_slots.emplace_back();
209 idx = static_cast<uint32_t>(m_material_slots.size() - 1);
210 }
211 else {
212 idx = m_material_free_slots.back();
213 m_material_free_slots.pop_back();
214 }
215
216 auto& slot = m_material_slots[idx];
217
218 // Temel Verileri Kopyala
219 slot.is_allocated = true;
220 slot.asset.guid = guid;
221 slot.asset.name = loaded->name;
222 slot.asset.mode = loaded->mode;
223 slot.asset.alpha_threshold = loaded->alpha_threshold;
224 slot.asset.base_color = loaded->base_color;
225 slot.asset.specular_strength = loaded->specular_strength;
226 slot.asset.shininess = loaded->shininess;
227 slot.asset.normal_strength = loaded->normal_strength;
228 slot.asset.texture_scale = loaded->texture_scale;
229
230 // Bağımlı Varlıkları Yükle (Doku ve Shader)
231 slot.asset.diffuse_texture_handle = (!loaded->diffuse_texture_guid.is_nil())
232 ? this->load_texture(loaded->diffuse_texture_guid) : handle<texture_asset>{};
233
234 slot.asset.normal_texture_handle = (!loaded->normal_texture_guid.is_nil())
235 ? this->load_texture(loaded->normal_texture_guid) : handle<texture_asset>{};
236
237 slot.asset.shader_handle = (!loaded->shader_guid.is_nil())
238 ? this->load_shader(loaded->shader_guid) : handle<shader_asset>{};
239
240 handle<material_asset> h{ idx, slot.generation };
241 m_material_map[guid] = h;
242
243 return h;
244 }
