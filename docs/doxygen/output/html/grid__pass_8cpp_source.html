27 // 1. Aria önbelleğinden ızgara shader tanıtıcısını al ve bağla

30
31 // 2. Kamera ve ters projeksiyon matrislerini hazırla
32 const glm::vec3& camPos = editorCtx.camera->position;
33 const glm::mat4 gridView = editorCtx.camera->getViewMatrix();
34
35 const float aspect = static_cast<float>(ctx.editorTarget.width) /
36 static_cast<float>(std::max(1, ctx.editorTarget.height));
37
38 const glm::mat4 gridProj = editorCtx.camera->getProjectionMatrix(aspect);
39 const glm::mat4 invViewProj = glm::inverse(gridProj * gridView);
40
41 // 3. Shader uniform değişkenlerini güncelle
42 const GLint invViewProjLoc = glGetUniformLocation(gridGLShaderID, "invViewProj");
43 if (invViewProjLoc != -1) {
44 glUniformMatrix4fv(invViewProjLoc, 1, GL_FALSE, glm::value_ptr(invViewProj));
45 }
46
47 const GLint camPosLoc = glGetUniformLocation(gridGLShaderID, "cameraPos");
48 if (camPosLoc != -1) {
49 glUniform3fv(camPosLoc, 1, glm::value_ptr(camPos));
50 }
51
52 // 4. Quad geometrisini çiz
53 ctx.grid.draw();
54
55 // 5. OpenGL durumlarını varsayılana geri yükle
56 glUseProgram(0);
57 glDisable(GL_BLEND);
58 glDepthMask(GL_TRUE);
59 glEnable(GL_DEPTH_TEST);
