123void gizmo_system::DrawWireSphere(const glm::mat4& transform, float radius, const glm::vec4& color)
124{
125 constexpr int segments = 32;
126 const float step = glm::two_pi<float>() / static_cast<float>(segments);
127
128 auto TransformPoint = [&](const glm::vec3& p) {
129 return glm::vec3(transform * glm::vec4(p, 1.0f));
130 };
131
132 for (int i = 0; i < segments; ++i)
133 {
134 const float a0 = static_cast<float>(i) * step;
135 const float a1 = static_cast<float>(i + 1) * step;
136
137 const float cos0 = std::cos(a0);
138 const float sin0 = std::sin(a0);
139 const float cos1 = std::cos(a1);
140 const float sin1 = std::sin(a1);
141
142 // XY Düzlemi
143 DrawLine(
144 TransformPoint(glm::vec3(cos0, sin0, 0.0f) * radius),
145 TransformPoint(glm::vec3(cos1, sin1, 0.0f) * radius),
146 color
147 );
148
149 // XZ Düzlemi
150 DrawLine(
151 TransformPoint(glm::vec3(cos0, 0.0f, sin0) * radius),
152 TransformPoint(glm::vec3(cos1, 0.0f, sin1) * radius),
153 color
154 );
155
156 // YZ Düzlemi
157 DrawLine(
158 TransformPoint(glm::vec3(0.0f, cos0, sin0) * radius),
159 TransformPoint(glm::vec3(0.0f, cos1, sin1) * radius),
160 color
161 );
162 }
163}

165void gizmo_system::Flush(const glm::mat4& view, const glm::mat4& projection, unsigned int shaderID)
166{
167 if (shaderID == 0 || m_LineVertices.empty())
168 {
169 m_LineVertices.clear();
170 return;
171 }
172
173 // 1. Programı bağla
174 glUseProgram(shaderID);
175
176 // 2. Uniform matrisini yükle
177 const int loc = glGetUniformLocation(shaderID, "u_ViewProj");
178 if (loc != -1) {
179 const glm::mat4 viewProj = projection * view;
180 glUniformMatrix4fv(loc, 1, GL_FALSE, &viewProj[0][0]);
181 }
182
183 // 3. Render durumlarını yapılandır
184 glEnable(GL_DEPTH_TEST);
185 glDepthMask(GL_FALSE);
186 glDepthFunc(GL_LEQUAL);
187 glLineWidth(2.0f);
188 glEnable(GL_POLYGON_OFFSET_LINE);
189 glPolygonOffset(-1.0f, -1.0f);
190
191 // 4. Veri yükleme
192 glBindBuffer(GL_ARRAY_BUFFER, m_VBO);
193 const size_t vertexCount = m_LineVertices.size();
194
195 if (vertexCount > m_AllocatedCapacity)
196 {
197 m_AllocatedCapacity = vertexCount * 2;
198 glBufferData(GL_ARRAY_BUFFER, m_AllocatedCapacity * sizeof(GizmoVertex), nullptr, GL_DYNAMIC_DRAW);
199 }
200
201 glBufferSubData(GL_ARRAY_BUFFER, 0, vertexCount * sizeof(GizmoVertex), m_LineVertices.data());
202
203 // 5. Çizim çağrısı
204 glBindVertexArray(m_VAO);
205 glDrawArrays(GL_LINES, 0, static_cast<GLsizei>(vertexCount));
206
207 // 6. Durumları varsayılana geri yükle
208 glBindVertexArray(0);
209 glBindBuffer(GL_ARRAY_BUFFER, 0);
210 glUseProgram(0);
211 glDisable(GL_POLYGON_OFFSET_LINE);
212 glDepthMask(GL_TRUE);
213 glDepthFunc(GL_LESS);
214 glLineWidth(1.0f);
215
216 m_LineVertices.clear();
217}
