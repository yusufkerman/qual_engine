17 unsigned int create_cubemap()
18 {
19 unsigned int texID = 0;
20 glGenTextures(1, &texID);
21 glBindTexture(GL_TEXTURE_CUBE_MAP, texID);
22
23 glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
24 glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
25 glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
26 glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
27 glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_WRAP_R, GL_CLAMP_TO_EDGE);
28
29 return texID;
30 }

34unsigned int cubemap_loader::load_faces(const std::vector<std::string>& faces)
35{
36 if (faces.size() != 6) {
37 std::cerr << "[CubemapLoader] Gecersiz yuz sayisi (6 yuz bekleniyordu): " << faces.size() << '\n';
38 return 0;
39 }
40
41 unsigned int textureID = create_cubemap();
42
43 int width = 0;
44 int height = 0;
45 int channels = 0;
46 stbi_set_flip_vertically_on_load(false);
47
48 for (unsigned int i = 0; i < 6; ++i)
49 {
50 unsigned char* data = stbi_load(faces[i].c_str(), &width, &height, &channels, 0);
51
52 if (!data)
53 {
54 std::cerr << "[CubemapLoader] Dosya yuklenemedi: " << faces[i] << '\n';
55 glDeleteTextures(1, &textureID);
56 return 0;
57 }
58
59 GLenum format = (channels == 4) ? GL_RGBA : GL_RGB;
60 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
61
62 glTexImage2D(
63 GL_TEXTURE_CUBE_MAP_POSITIVE_X + i,
64 0,
65 format,
66 width,
67 height,
68 0,
69 format,
70 GL_UNSIGNED_BYTE,
71 data
72 );
73
74 stbi_image_free(data);
75 }
76

80unsigned int cubemap_loader::load_atlas(const std::string& path)
81{
82 int width = 0;
83 int height = 0;
84 int channels = 0;
85 stbi_set_flip_vertically_on_load(false);
86
87 unsigned char* data = stbi_load(path.c_str(), &width, &height, &channels, 0);
88
89 if (!data)
90 {
91 std::cerr << "[CubemapLoader] Atlas dosyasi bulunamadi: " << path << '\n';
92 return 0;
93 }
94
95 const int faceW = width / 4;
96 const int faceH = height / 3;
97
98 if (faceW <= 0 || faceH <= 0)
99 {
100 std::cerr << "[CubemapLoader] Gecersiz atlas cozunurlugu: " << width << "x" << height << '\n';
101 stbi_image_free(data);
102 return 0;
103 }
104
105 unsigned int texID = create_cubemap();
106 const GLenum format = (channels == 4) ? GL_RGBA : GL_RGB;
107
108 auto uploadFace = [&](int fx, int fy, GLenum target)
109 {
110 std::vector<unsigned char> faceData(faceW * faceH * channels);
111
112 // Satır satır bellek kopyalama ile alt dikdörtgeni ayıkla
113 for (int row = 0; row < faceH; ++row)
114 {
115 const unsigned char* src = data + (((fy * faceH + row) * width + (fx * faceW)) * channels);
116 unsigned char* dst = faceData.data() + (row * faceW * channels);
117 std::memcpy(dst, src, faceW * channels);
118 }
119
120 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
121 glTexImage2D(target, 0, format, faceW, faceH, 0, format, GL_UNSIGNED_BYTE, faceData.data());
122 };
123
124 // Standart 4x3 Haç (Cross) yerleşimi:
125 // [+Y]
126 // [-X] [+Z] [+X] [-Z]
127 // [-Y]
128 uploadFace(1, 0, GL_TEXTURE_CUBE_MAP_POSITIVE_Y); // Üst Orta (+Y)
129 uploadFace(0, 1, GL_TEXTURE_CUBE_MAP_NEGATIVE_X); // Orta Sol (-X)
130 uploadFace(1, 1, GL_TEXTURE_CUBE_MAP_POSITIVE_Z); // Orta (+Z / Ön)
131 uploadFace(2, 1, GL_TEXTURE_CUBE_MAP_POSITIVE_X); // Orta Sağ (+X)
132 uploadFace(3, 1, GL_TEXTURE_CUBE_MAP_NEGATIVE_Z); // En Sağ (-Z / Arka)
133 uploadFace(1, 2, GL_TEXTURE_CUBE_MAP_NEGATIVE_Y); // Alt Orta (-Y)
134
135 stbi_image_free(data);
136 return texID;
137}
