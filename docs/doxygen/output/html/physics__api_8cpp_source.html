71glm::mat4 physics_api::get_jolt_transform(entity e)
72{
73 const auto* rb = m_reg->get_component<rigidbody_component>(e);
74
75 if (!rb || rb->body_id.IsInvalid())
76 {
77 return glm::mat4(1.0f);
78 }
79
80 auto& bodyInterface = m_phys->get_jolt_system()->GetBodyInterface();
81
82 JPH::RVec3 joltPos;
83 JPH::Quat joltRot;
84 bodyInterface.GetPositionAndRotation(rb->body_id, joltPos, joltRot);
85
86 const glm::vec3 pos = glm::vec3(joltPos.GetX(), joltPos.GetY(), joltPos.GetZ());
87
88 glm::quat rot;
89 rot.w = joltRot.GetW();
90 rot.x = joltRot.GetX();
91 rot.y = joltRot.GetY();
92 rot.z = joltRot.GetZ();
93
94 // Dönüşüm işlemleri doğrudan quaternion üzerinden uygulanır.
95 glm::mat4 modelMatrix = glm::translate(glm::mat4(1.0f), pos);
96 modelMatrix = modelMatrix * glm::mat4_cast(rot);
97
98 return modelMatrix;
99}

102{
