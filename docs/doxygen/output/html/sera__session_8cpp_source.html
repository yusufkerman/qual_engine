215 validation_report sera_session::load(const std::string& filepath, scene& out_scene)
216 {
217 validation_report report;
218
219 std::ifstream file(filepath);
220 if (!file.is_open())
221 {
222 report.add_critical("Sahne dosyasi diskten okunamadi: " + filepath);
223 return report;
224 }
225
226 json scene_json;
227 try
228 {
229 file >> scene_json;
230 }
231 catch (const json::parse_error& e)
232 {
233 file.close();
234 report.add_critical("JSON format hatasi: " + std::string(e.what()));
235 return report;
236 }
237 file.close();
238
239 out_scene.new_scene();
240 m_entity_registry.clear();
241 registry& reg = out_scene.get_registry();
242
243 std::unordered_map<std::string, entity> uuid_to_runtime_entity_map;
244
245 if (scene_json.contains("entities"))
246 {
247 // 1. Aşama: Varlık tanıtıcılarını oluştur ve UUID bağlantılarını kur
248 for (const auto& entity_json : scene_json["entities"])
249 {
250 std::string uuid = entity_json.value("uuid", "");
251 if (uuid.empty())
252 {
253 uuid = sera_entity_registry::generate_uuid();
254 }
255
256 const entity new_e = out_scene.create_entity();
257 m_entity_registry.register_link(new_e, uuid);
258 uuid_to_runtime_entity_map[uuid] = new_e;
259 }
260
261 // 2. Aşama: Bileşenleri ilgili varlıklara deserialize et
262 for (const auto& entity_json : scene_json["entities"])
263 {
264 const std::string uuid = entity_json["uuid"];
265 const entity current_e = uuid_to_runtime_entity_map[uuid];
266
267 if (!entity_json.contains("components"))
268 {
269 continue;
270 }
271 const auto& comps_json = entity_json["components"];
272
273 // Transform Component
274 if (comps_json.contains("transform_component"))
275 {
276 transform_component comp{};
277 deserialize_visitor visitor{ comps_json["transform_component"] };
278 sera_reflect_component(comp, visitor);
279 reg.add_component<transform_component>(current_e, comp);
280 }
281
282 // Name Component
283 if (comps_json.contains("name_component"))
284 {
285 name_component comp{};
286 deserialize_visitor visitor{ comps_json["name_component"] };
287 sera_reflect_component(comp, visitor);
288 reg.add_component<name_component>(current_e, comp);
289 }
290
291 // Mesh Component (Aria Model Handle)
292 if (comps_json.contains("mesh_component"))
293 {
294 mesh_component comp{};
295
296 const std::string guid_str = comps_json["mesh_component"].value("model_guid", "");
297 const auto parsed = uuids::uuid::from_string(guid_str);
298
299 if (parsed)
300 {
301 comp.model_handle = m_asset_resolver.to_handle<aria::model_asset>(*parsed);
302 }
303
304 comp.sub_mesh_index = comps_json["mesh_component"].value("sub_mesh_index", size_t{ 0 });
305 comp.source = static_cast<mesh_source_type>(comps_json["mesh_component"].value("source", 0));
306
307 reg.add_component<mesh_component>(current_e, comp);
308 }
309
310 // Material Component (Aria Material Handle)
311 if (comps_json.contains("material_component"))
312 {
313 material_component comp{};
314
315 const std::string guid_str = comps_json["material_component"].value("material_guid", "");
316 const auto parsed = uuids::uuid::from_string(guid_str);
317
318 if (parsed)
319 {
320 comp.material_handle = m_asset_resolver.to_handle<aria::material_asset>(*parsed);
321 }
322
323 reg.add_component<material_component>(current_e, comp);
324 }
325
326 // Camera Component
327 if (comps_json.contains("camera_component"))
328 {
329 camera_component comp{};
330 deserialize_visitor visitor{ comps_json["camera_component"] };
331 sera_reflect_component(comp, visitor);
332 reg.add_component<camera_component>(current_e, comp);
333 }
334
335 // Light Component
336 if (comps_json.contains("light_component"))
337 {
338 light_component comp{};
339 deserialize_visitor visitor{ comps_json["light_component"] };
340 sera_reflect_component(comp, visitor);
341 reg.add_component<light_component>(current_e, comp);
342 }
343
344 // Bounds Component
345 if (comps_json.contains("bounds_component"))
346 {
347 bounds_component comp{};
348 deserialize_visitor visitor{ comps_json["bounds_component"] };
349 sera_reflect_component(comp, visitor);
350 reg.add_component<bounds_component>(current_e, comp);
351 }
352
353 // Rigidbody Component
354 if (comps_json.contains("rigidbody_component"))
355 {
356 rigidbody_component comp{};
357 deserialize_visitor visitor{ comps_json["rigidbody_component"] };
358 sera_reflect_component(comp, visitor);
359 reg.add_component<rigidbody_component>(current_e, comp);
360 }
361
362 // Collider Component
363 if (comps_json.contains("collider_component"))
364 {
365 collider_component comp{};
366 deserialize_visitor visitor{ comps_json["collider_component"] };
367 sera_reflect_component(comp, visitor);
368 reg.add_component<collider_component>(current_e, comp);
369 }
370
371 // Native Script Component
372 if (comps_json.contains("native_script_component"))
373 {
374 const auto& sc = comps_json["native_script_component"];
375 const std::string name = sc.value("script_name", "None");
376
377 if (!name.empty() && name != "None")
378 {
379 auto& comp = reg.add_component<native_script_component>(current_e);
380 comp.script_name = name;
381 script_factory::instantiate_script(name, current_e, reg);
382 }
383 }
384 }
385 }
386
387 // Aktif kameranın geri yüklenmesi
388 if (scene_json.contains("active_camera_uuid"))
389 {
390 const std::string cam_uuid = scene_json["active_camera_uuid"];
391 if (!cam_uuid.empty())
392 {
393 const auto it = uuid_to_runtime_entity_map.find(cam_uuid);
394 if (it != uuid_to_runtime_entity_map.end())
395 {
396 out_scene.set_active_camera(it->second);
397 }
398 }
399 }
400
401 out_scene.update_light_batch();
402 report.add_info("Sahne ve bagli Aria varlik referanslari basariyla geri yuklendi.");
403 return report;
404 }
