22void editor_pass::render(EngineContext& ctx, EditorContext& editorCtx)
23{
24 if (!editorCtx.camera) {
25 return;
26 }
27
28 // 1. Hedef Framebuffer ve Viewport yapılandırması
29 glBindFramebuffer(GL_FRAMEBUFFER, ctx.editorTarget.fbo);
30 glViewport(0, 0, ctx.editorTarget.width, ctx.editorTarget.height);
31
32 glEnable(GL_DEPTH_TEST);
33 glDepthMask(GL_TRUE);
34 glClearColor(0.08f, 0.08f, 0.12f, 1.0f);
35 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
36
37 // 2. Kamera matrislerini hesapla (sıfıra bölme koruması ile)
38 const float aspect = static_cast<float>(ctx.editorTarget.width) /
39 static_cast<float>(std::max(1, ctx.editorTarget.height));
40
41 const glm::mat4 editorView = editorCtx.camera->getViewMatrix();
42 const glm::mat4 editorProj = editorCtx.camera->getProjectionMatrix(aspect);
43
44 // 3. Gökyüzü (Skybox) render geçişi
45 skybox_pass::render(ctx, editorView, editorProj);
46
47 // 4. Ana sahne geometrisi ve ışıklandırma
48 ctx.renderSystem.render(
49 ctx.session.editor_scene().get_registry(),
50 ctx.session.editor_scene().get_light_batch(),
51 ctx,
52 editorView,
53 editorProj
54 );
55
56 // 5. Seçili varlık için sınır kutusu / çarpıştırıcı (Collider Gizmo) çizimi
57 auto& reg = ctx.session.editor_scene().get_registry();
58 const int selectedID = ctx.vestaSys.selection().get_first_selected_entity();
59
60 if (selectedID != -1)
61 {
62 const auto selectedEntity = static_cast<entity>(selectedID);
63
64 const auto* transform = reg.get_component<transform_component>(selectedEntity);
65 const auto* collider = reg.get_component<collider_component>(selectedEntity);
66 const auto* rb = reg.get_component<rigidbody_component>(selectedEntity);
67
68 if (transform && collider)
69 {
70 glm::mat4 modelMatrix = glm::mat4(1.0f);
71 bool physicsActive = false;
72
73 // Simülasyon modunda dönüşümü fizik motorundan (Jolt) al
74 if (rb && !rb->body_id.IsInvalid())
75 {
76 modelMatrix = ctx.physicsSystem.api().get_jolt_transform(selectedEntity);
77 physicsActive = true;
78 }
79
80 // Editör modunda yerel dönüşüm bileşenini kullan
81 if (!physicsActive)
82 {
83 modelMatrix = glm::translate(modelMatrix, transform->position);
84 modelMatrix = glm::rotate(modelMatrix, transform->rotation.x, glm::vec3(1.0f, 0.0f, 0.0f));
85 modelMatrix = glm::rotate(modelMatrix, transform->rotation.y, glm::vec3(0.0f, 1.0f, 0.0f));
86 modelMatrix = glm::rotate(modelMatrix, transform->rotation.z, glm::vec3(0.0f, 0.0f, 1.0f));
87 }
88
89 constexpr glm::vec4 gizmoColor(0.0f, 1.0f, 0.0f, 1.0f);
90
91 switch (collider->type)
92 {
93 case collider_type::box:
94 ctx.gizmoSystem.DrawWireCube(
95 modelMatrix,
96 collider->extents,
97 gizmoColor
98 );
99 break;
100
101 case collider_type::sphere:
102 ctx.gizmoSystem.DrawWireSphere(
103 modelMatrix,
104 collider->extents.x,
105 gizmoColor
106 );
107 break;
108
109 default:
110 break;
111 }
112 }
113 }
114
115 // 6. Aria varlık önbelleğinden gizmo shader programını al ve çizgileri ekrana boşalt
116 if (ctx.aria_system)
117 {
118 const unsigned int gizmoGLShaderID = ctx.aria_system->get_shader_id(ctx.gizmoShaderHandle);
119 ctx.gizmoSystem.Flush(editorView, editorProj, gizmoGLShaderID);
120 }
121}
