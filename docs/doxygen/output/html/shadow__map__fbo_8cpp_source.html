15bool shadow_map_fbo::init(unsigned int width, unsigned int height)
16{
17 // Yeniden başlatma durumunda eski kaynakları serbest bırak
18 destroy();
19
20 m_Width = width;
21 m_Height = height;
22
23 // 1. Framebuffer oluştur
24 glGenFramebuffers(1, &m_FBO);
25
26 // 2. Yalnızca derinlik verisi tutacak olan dokuyu yapılandır
27 glGenTextures(1, &m_DepthTexture);
28 glBindTexture(GL_TEXTURE_2D, m_DepthTexture);
29
30 glTexImage2D(
31 GL_TEXTURE_2D,
32 0,
33 GL_DEPTH_COMPONENT,
34 static_cast<GLsizei>(m_Width),
35 static_cast<GLsizei>(m_Height),
36 0,
37 GL_DEPTH_COMPONENT,
38 GL_FLOAT,
39 nullptr
40 );
41
42 // Gölge filtreleme parametreleri
43 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
44 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
45
46 // Işık frustum'ı dışındaki koordinatların gölgede kalmaması için sınır rengini 1.0 (beyaz/uzak) yap
47 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_BORDER);
48 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_BORDER);
49 constexpr float borderColor[] = { 1.0f, 1.0f, 1.0f, 1.0f };
50 glTexParameterfv(GL_TEXTURE_2D, GL_TEXTURE_BORDER_COLOR, borderColor);
51
52 // 3. Dokuyu Framebuffer'ın derinlik noktasına bağla
53 glBindFramebuffer(GL_FRAMEBUFFER, m_FBO);
54 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT, GL_TEXTURE_2D, m_DepthTexture, 0);
55
56 // Renk tamponu yazım ve okuma işlemlerini devre dışı bırak
57 glDrawBuffer(GL_NONE);
58 glReadBuffer(GL_NONE);
59
60 // Doğrulama kontrolü
61 const GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
62 if (status != GL_FRAMEBUFFER_COMPLETE)
63 {
64 std::cerr << "[ShadowMapFBO] Hata: Framebuffer eksik veya gecersiz! (Durum: " << status << ")\n";
65 glBindFramebuffer(GL_FRAMEBUFFER, 0);
66 destroy();
67 return false;
68 }
69
70 glBindFramebuffer(GL_FRAMEBUFFER, 0);
71 return true;
72}
