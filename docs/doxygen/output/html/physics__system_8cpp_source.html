19bool physics_system::init(scene& s)
20{
21 // Fizik komut arayüzünü (API) oluştur ve mevcut sahnenin kayıt defterine bağla
22 m_api = std::make_unique<physics_api>(this, &s.get_registry());
23
24 // Jolt bellek ayırıcı (allocator) altyapısını ilklendir
25 JPH::RegisterDefaultAllocator();
26
27 // Jolt nesne üretimi için genel fabrika (Factory) örneğini ayarla
28 JPH::Factory::sInstance = new JPH::Factory();
29
30 // Jolt temel çarpışma veri tiplerini sisteme tanıt
31 JPH::RegisterTypes();
32
33 // Simülasyon döngüsündeki geçici hesaplamalar için bellek ayırıcı (10 MB ayrılmıştır)
34 mTempAllocator = std::make_unique<JPH::TempAllocatorImpl>(10 * 1024 * 1024);
35
36 // Fizik hesaplamalarını paralelleştirmek için çoklu iş parçacığı havuzu (Mevcut donanım sınırının 1 eksiği)
37 mJobSystem = std::make_unique<JPH::JobSystemThreadPool>(
38 JPH::cMaxPhysicsJobs,
39 JPH::cMaxPhysicsBarriers,
40 std::thread::hardware_concurrency() - 1
41 );
42
43 // Ana fizik sistemini ayrılan sınırlar ve kurulan filtreler eşliğinde başlat
44 mPhysicsSystem = std::make_unique<JPH::PhysicsSystem>();
45 mPhysicsSystem->Init(
46 mMaxBodies,
47 mNumBodyMutexes,
48 mMaxBodyPairs,
49 mMaxContactConstraints,
50 mBPInterface,
51 mObjectVsBPFilter,
52 mObjectVsObjectFilter
53 );
54
55 return true;
56}
