68float CalculateShadow(vec4 fragPosLightSpace, vec3 N, vec3 L)
69{
70 vec3 projCoords = fragPosLightSpace.xyz / fragPosLightSpace.w;
71 projCoords = projCoords * 0.5 + 0.5;
72
73 if(projCoords.z > 1.0)
74 return 0.0;
75
76 float bias = max(0.0015 * (1.0 - dot(N,L)), 0.0003);
77
78 vec2 texelSize = 1.0 / textureSize(shadowMap,0);
79
80 float shadow = 0.0;
81
82 for(int x=-1;x<=1;x++)
83 {
84 for(int y=-1;y<=1;y++)
85 {
86 float depth = texture(
87 shadowMap,
88 projCoords.xy + vec2(x,y)*texelSize
89 ).r;
90
91 shadow +=
92 (projCoords.z-bias>depth)
93 ?1.0:0.0;
94 }
95 }
96
97 shadow/=9.0;
98
99 return shadow;
100}
101void main()
102{
103
104 vec2 tiledUV = vUV * u_TextureScale;
105
106 vec4 texColor = useTexture ? texture(diffuseTex, tiledUV) : vec4(baseColor, 1.0);
107 if (u_RenderMode == 1)
108 {
109 if (texColor.a < u_AlphaThreshold)
110 {
111 discard; // Bu komut çalıştığı an shader aşağıya devam etmez, pikseller şeffaf olur!
112 }
113 }
114
115 vec3 albedo = texColor.rgb;
116 vec3 N = normalize(vNormal);
117
118 if (useNormalMap)
119 {
120 vec3 tangentNormal = texture(normalTex, tiledUV).rgb * 2.0 - 1.0;
121 tangentNormal.xy *= normalStrength;
122 N = normalize(vTBN * normalize(tangentNormal));
123 }
124
125
126 vec3 result = vec3(0.0);
127 float ambient = 0.15;
128
129 result += albedo * ambient;
130
131 // Pikselden kameraya doğru giden ortak görüş vektörü (V)
132 vec3 V = normalize(viewPos - vWorldPos);
133
134 for (int i = 0; i < lightCount && i < 64; i++)
135 {
136 Light l = lights[i];
137
138 vec3 L;
139 float attenuation = 1.0;
140
141 if (l.type == 0) // DIRECTIONAL
142 {
143 L = normalize(-l.direction);
144 }
145 else if (l.type == 1) // POINT
146 {
147 vec3 toLight = l.position - vWorldPos;
148 float dist = length(toLight);
149 L = (dist > 0.0001) ? (toLight / dist) : vec3(0.0);
150
151 attenuation = 1.0 / (1.0 + (dist / l.range) * (dist / l.range));
152 }
153 else // SPOT
154 {
155 vec3 toLight = l.position - vWorldPos;
156 float dist = length(toLight);
157 L = toLight / dist;
158
159 vec3 lightDir = normalize(l.direction);
160 float theta = dot(-lightDir, L);
161
162 float inner = cos(radians(l.inner_cone));
163 float outer = cos(radians(l.outer_cone));
164
165 float spot = smoothstep(outer, inner, theta);
166
167 float distanceAtt = 1.0 / (1.0 + (dist * dist) / (l.range * l.range));
168 attenuation *= distanceAtt * spot;
169 }
170
171 // 1. Diffuse (Dağınık Aydınlatma)
172 float diff = max(dot(N, L), 0.0);
173 vec3 diffuse = albedo * l.color * l.intensity * diff * attenuation;
174
175 // Blinn-Phong Specular (Parlama) Hesaplaması
176 vec3 specular = vec3(0.0);
177 if (diff > 0.0) // Eğer yüzey ışık alıyorsa parlayabilir
178 {
179 // Işık vektörü (L) ile Görüş vektörünün (V) tam ortasındaki vektör (Halfway Vector)
180 vec3 H = normalize(L + V);
181
182 // Parlama keskinliği (Shininnes/Shininess Exponent).
183 // 32.0 plastik gibidir, 64-128 metallere doğru gider. Şimdilik sabit veriyoruz.
184 float specPower = pow(max(dot(N, H), 0.0), shininess);
185
186 // Işığın kendi rengi ve şiddetiyle parlamayı çarpıyoruz
187 specular = l.color * l.intensity * specPower * attenuation * specularStrength;
188 }
189
190 float shadow = 0.0;
191 if (l.type == 0)
192 {
193 shadow = CalculateShadow(vFragPosLightSpace, N, L);
194 }
195
196 // Nihai sonuca diffuse'un yanına parlama (specular) efektini de ekliyoruz
197 result += (1.0 - shadow) * (diffuse + specular);
198 }
199
200 // ============================================================================
201 // EXPONENTIAL FOG
202 // ============================================================================
203 if (u_FogEnabled)
204 {
205 // Pikselin kameraya olan gerçek dünyadaki mesafesi
206 float distance = length(viewPos - vWorldPos);
207
208 // Exponential (Üstel) Sis Formülü: e^(-density * distance)
209 float fogFactor = exp(-u_FogDensity * distance);
210 fogFactor = clamp(fogFactor, 0.0, 1.0);
211
212 // Işıklandırılmış rengi sis rengi ile mix et
213 result = mix(u_FogColor, result, fogFactor);
214 }
215 // ============================================================================
216
217 float finalAlpha = (u_RenderMode == 2) ? texColor.a : 1.0;
218
219 FragColor = vec4(result, finalAlpha);
220}
