18 void inspector_panel::render()
19 {
20 if (!m_is_open)
21 return;
22
23 m_vm->update();
24
25 ImGui::Begin(m_title.c_str(), &m_is_open);
26
27 switch (m_vm->get_mode())
28 {
29 // ========================================================
30 // 1. SEÇİLİ ÖĞE YOK (NOTHING SELECTED)
31 // ========================================================
32 case inspector_mode::none:
33 {
34 ImGui::TextDisabled("Nothing selected.");
35 break;
36 }
37
38 // ========================================================
39 // 2. SAHNE NESNESİ DENETİMİ (ENTITY INSPECTOR)
40 // ========================================================
41 case inspector_mode::entity:
42 {
43 if (ImGui::Button("Add Component"))
44 {
45 ImGui::OpenPopup("AddComponentPopup");
46 }
47
48 if (ImGui::BeginPopup("AddComponentPopup"))
49 {
50 for (const auto& component : m_vm->get_addable_components())
51 {
52 if (ImGui::MenuItem(component.info->display_name.c_str()))
53 {
54 m_vm->request_add_component(component.hash);
55 ImGui::CloseCurrentPopup();
56 }
57 }
58 ImGui::EndPopup();
59 }
60
61 ImGui::Separator();
62
63 for (const auto& component : m_vm->get_components())
64 {
65 ImGui::PushID(static_cast<int>(component.hash));
66
67 // Başlık çubuğu ve açılır/kapanır sekme kontrolü
68 bool opened = ImGui::CollapsingHeader(
69 component.display_name.c_str(),
70 component.default_open ? ImGuiTreeNodeFlags_DefaultOpen : 0
71 );
72
73 if (component.removable)
74 {
75 // Silme butonunu başlığın sağ kenarına hizala
76 ImGui::SameLine(ImGui::GetWindowWidth() - 80.0f);
77 if (ImGui::SmallButton("Remove"))
78 {
79 m_vm->request_remove_component(component.hash);
80 ImGui::PopID();
81 continue; // Silinen bileşenin kalan arayüzünü bu kare çizmeden atla
82 }
83 }
84
85 if (opened && component.ui && component.ui->render_fn)
86 {
87 component.ui->render_fn(
88 m_vm->get_selected_entity(),
89 m_vm->get_registry()
90 );
91 }
92
93 ImGui::PopID();
94 }
95
96 break;
97 }
98
99 // ========================================================
100 // 3. DOSYA/VARLIK DENETİMİ (ASSET INSPECTOR)
101 // ========================================================
102 case inspector_mode::asset:
103 {
104 auto asset = m_vm->get_asset_data();
105
106 if (!asset.ui || !asset.ui->render_fn)
107 {
108 ImGui::TextDisabled("No inspector registered for this asset type.");
109 break;
110 }
111
112 asset.ui->render_fn(asset.asset_path);
113 break;
114 }
115 }
116
117 ImGui::End();
118 }
119
120} // namespace vesta
