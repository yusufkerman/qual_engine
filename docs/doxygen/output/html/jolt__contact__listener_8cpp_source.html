25 void jolt_contact_listener::OnContactAdded(
26 const JPH::Body& inBody1,
27 const JPH::Body& inBody2,
28 const JPH::ContactManifold& inManifold,
29 JPH::ContactSettings& ioSettings)
30 {
31 const entity ent1 = static_cast<entity>(inBody1.GetUserData());
32 const entity ent2 = static_cast<entity>(inBody2.GetUserData());
33
34 if (ent1 == 0 || ent2 == 0)
35 {
36 return;
37 }
38
39 const uint32_t b1 = inBody1.GetID().GetIndexAndSequenceNumber();
40 const uint32_t b2 = inBody2.GetID().GetIndexAndSequenceNumber();
41
42 // Çifti önbelleğe al (OnContactRemoved için)
43 {
44 std::lock_guard<std::mutex> lock(m_cache_mutex);
45 m_active_pairs[{b1, b2}] = { ent1, ent2, ioSettings.mIsSensor };
46 }
47
48 if (!m_queue)
49 {
50 return;
51 }
52
53 pending_collision_event ev;
54 ev.entity_a = ent1;
55 ev.entity_b = ent2;
56 ev.type = collision_event_type::enter;
57 ev.is_sensor = ioSettings.mIsSensor;
58
59 // Temas noktası ve normali
60 const JPH::RVec3 pt = inManifold.mBaseOffset;
61 ev.point = glm::vec3(pt.GetX(), pt.GetY(), pt.GetZ());
62 ev.normal = glm::vec3(inManifold.mWorldSpaceNormal.GetX(),
63 inManifold.mWorldSpaceNormal.GetY(),
64 inManifold.mWorldSpaceNormal.GetZ());
65 ev.impulse = inManifold.mPenetrationDepth;
66
67 m_queue->push_event(ev);
68 }

100 void jolt_contact_listener::OnContactRemoved(const JPH::SubShapeIDPair& inSubShapePair)
101 {
102 const uint32_t b1 = inSubShapePair.GetBody1ID().GetIndexAndSequenceNumber();
103 const uint32_t b2 = inSubShapePair.GetBody2ID().GetIndexAndSequenceNumber();
104
105 entity_pair_info info;
106 bool found = false;
107
108 {
109 std::lock_guard<std::mutex> lock(m_cache_mutex);
110 auto it = m_active_pairs.find({ b1, b2 });
111 if (it != m_active_pairs.end())
112 {
113 info = it->second;
114 found = true;
115 m_active_pairs.erase(it);
116 }
117 }
118
119 if (found && m_queue)
120 {
121 pending_collision_event ev;
122 ev.entity_a = info.entity_a;
123 ev.entity_b = info.entity_b;
124 ev.type = collision_event_type::exit;
125 ev.is_sensor = info.is_sensor;
126 m_queue->push_event(ev);
127 }
128 }
