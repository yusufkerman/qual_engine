3out vec4 FragColor;

5in vec2 uv;
6
7uniform mat4 invViewProj;
8uniform mat4 viewProj; // 🔥 DERİNLİK HESABI İÇİN EKLENDİ
9uniform vec3 cameraPos;
10
11// Kalınlığı katman ölçeğine göre dengeleyen fonksiyon
12float analyticalGrid(vec2 p, float scale, float dist)
13{
14 vec2 coord = p / scale;
15
16 // 🔥 ANA ÇİZGİLERİN KALINLAŞMASINI ENGELLEYEN FORMÜL:
17 // Kalınlığı scale değerine bölerek, büyük gridlerin (scale büyüdükçe)
18 // devasa kalınlıklara ulaşmasını engelliyoruz.
19 float baseThickness = 0.04;
20 float thickness = baseThickness * (1.0 + pow(dist, 1.1) * 0.02);
21
22 // Büyük ölçekli gridlerin çizgilerini alt katmana göre daha zarif yapar
23 thickness = min(thickness, scale * 0.05);
24
25 vec2 gridReal = abs(fract(coord - 0.5) - 0.5) / (thickness / scale);
26 vec2 line = smoothstep(1.0, 0.0, gridReal);
27
28 return max(line.x, line.y);
29}
30
31void main()
32{
33 vec2 ndc = uv * 2.0 - 1.0;
34
35 vec4 near = invViewProj * vec4(ndc, 0.0, 1.0);
36 vec4 far = invViewProj * vec4(ndc, 1.0, 1.0);
37
38 near /= near.w;
39 far /= far.w;
40
41 vec3 dir = normalize(far.xyz - near.xyz);
42
43 float t = -near.y / dir.y;
44 if (t < 0.0) discard;
45
46 vec3 worldPos = near.xyz + dir * t;
47 vec2 xz = worldPos.xz;
48
49 float dist = length(worldPos - cameraPos);
50
51 // 🔥 1. DİĞER MESHLERİN ARKASINDA RENDER OLMA FIXI (Depth Yazımı)
52 // Elle hesapladığımız dünya koordinatını tekrar View-Projection ile çarpıp
53 // ekran kartının derinlik testine (Z-Buffer) sokuyoruz.
54 vec4 clipPos = viewProj * vec4(worldPos, 1.0);
55 gl_FragDepth = clipPos.z / clipPos.w;
56
57 // 5 tabanlı logaritmik ölçeklendirme
58 float logDist = log2(max(dist, 1.0)) / log2(3.0);
59 float lod1 = floor(logDist);
60 float lod2 = lod1 + 1.0;
61
62 float scale1 = pow(3.0, lod1);
63 float scale2 = pow(3.0, lod2);
64
65 float g1 = analyticalGrid(xz, scale1, dist);
66 float g2 = analyticalGrid(xz, scale2, dist);
67
68 float f = fract(logDist);
69 float blend = smoothstep(0.0, 1.0, f);
70
71 float grid = mix(g1, g2, blend);
72
73 float horizonFade = 1.0 - smoothstep(1000.0, 3000.0, dist);
74 grid *= horizonFade;
75
76 grid = clamp(grid, 0.0, 1.0);
77
78 FragColor = vec4(vec3(0.35), grid);
79}
