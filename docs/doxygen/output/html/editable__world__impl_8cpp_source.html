31 void editable_world_impl::destroy_entity(entity e) {
32 // 1. Ebeveyn ve çocuk ilişkilerini hiyerarşik olarak çöz
33 if (m_registry.has_component<hierarchy_component>(e)) {
34 set_entity_parent(e, 0);
35
36 auto* hc = m_registry.get_component<hierarchy_component>(e);
37 if (hc) {
38 entity current_child = hc->first_child;
39 while (current_child != 0) {
40 auto* child_hc = m_registry.get_component<hierarchy_component>(current_child);
41 entity next_sibling = child_hc ? child_hc->next_sibling : 0;
42
43 set_entity_parent(current_child, 0);
44 current_child = next_sibling;
45 }
46 }
47 }
48
49 // 2. Kayıtlı tüm bileşen silme kancalarını (Observer dispatch) ve yıkım fonksiyonlarını tetikle
50 const auto& operation_registry = vesta::component_operation_registry::get_registry();
51
52 for (const auto& [hash, info] : operation_registry) {
53 if (has_component_by_hash(e, hash)) {
54 remove_component_by_hash(e, hash);
55 }
56 }
57
58 // 3. Varlığı ECS kayıt defterinden tamamen kaldır
59 m_registry.remove_entity(e);
60 mark_scene_dirty();
61 }

63 void editable_world_impl::set_entity_parent(entity child, entity parent) {
64 if (child == parent || child == 0) {
65 return;
66 }
67
68 // Döngüsel hiyerarşi engelleme: Hedef parent, çocuğun soyundan geliyor mu?
69 if (parent != 0 && m_registry.has_component<hierarchy_component>(parent)) {
70 entity ancestor = parent;
71 while (ancestor != 0) {
72 if (ancestor == child) {
73 return; // Döngüsel ilişki saptandı, atamayı iptal et
74 }
75 auto* anc_hc = m_registry.get_component<hierarchy_component>(ancestor);
76 ancestor = anc_hc ? anc_hc->parent : 0;
77 }
78 }
79
80 glm::mat4 old_world = ecs_transform::get_world_matrix(child, m_registry);
81
82 // 1. Child önceden bir parent'a bağlıysa, eski ebeveynin bağlı listesinden çıkar
83 if (m_registry.has_component<hierarchy_component>(child)) {
84 auto* child_hc = m_registry.get_component<hierarchy_component>(child);
85 if (child_hc && child_hc->parent != 0) {
86 auto* old_parent_hc = m_registry.get_component<hierarchy_component>(child_hc->parent);
87
88 if (child_hc->prev_sibling != 0) {
89 auto* prev_hc = m_registry.get_component<hierarchy_component>(child_hc->prev_sibling);
90 if (prev_hc) {
91 prev_hc->next_sibling = child_hc->next_sibling;
92 }
93 }
94 else if (old_parent_hc) {
95 old_parent_hc->first_child = child_hc->next_sibling;
96 }
97
98 if (child_hc->next_sibling != 0) {
99 auto* next_hc = m_registry.get_component<hierarchy_component>(child_hc->next_sibling);
100 if (next_hc) {
101 next_hc->prev_sibling = child_hc->prev_sibling;
102 }
103 }
104
105 child_hc->parent = 0;
106 child_hc->next_sibling = 0;
107 child_hc->prev_sibling = 0;
108 }
109 }
110
111 // 2. Yeni atanan parent kök dizin ise (parent == 0) dünya koordinatlarını koru ve çık
112 if (parent == 0) {
113 auto* t = m_registry.get_component<transform_component>(child);
114 if (t) {
115 ecs_transform::apply_world_matrix_as_local(*t, old_world);
116 }
117 mark_scene_dirty();
118 return;
119 }
120
121 // 3. Parent ve Child için hiyerarşi bileşenlerini doğrula
122 if (!m_registry.has_component<hierarchy_component>(parent)) {
123 m_registry.add_component<hierarchy_component>(parent);
124 }
125 if (!m_registry.has_component<hierarchy_component>(child)) {
126 m_registry.add_component<hierarchy_component>(child);
127 }
128
129 auto* child_hc = m_registry.get_component<hierarchy_component>(child);
130 auto* parent_hc = m_registry.get_component<hierarchy_component>(parent);
131
132 // 4. Yeni parent'ın çocuk listesinin en başına yerleştir (Prepend)
133 child_hc->parent = parent;
134 child_hc->next_sibling = parent_hc->first_child;
135 child_hc->prev_sibling = 0;
136
137 if (parent_hc->first_child != 0) {
138 auto* old_first_child_hc = m_registry.get_component<hierarchy_component>(parent_hc->first_child);
139 if (old_first_child_hc) {
140 old_first_child_hc->prev_sibling = child;
141 }
142 }
143
144 parent_hc->first_child = child;
145
146 // 5. Dünya konumunu korumak için yeni ebeveyne göre yerel dönüşüm matrisini hesapla
147 glm::mat4 new_parent_world = ecs_transform::get_world_matrix(parent, m_registry);
148 glm::mat4 new_local = glm::inverse(new_parent_world) * old_world;
149
150 auto* t = m_registry.get_component<transform_component>(child);
151 if (t) {
152 ecs_transform::apply_world_matrix_as_local(*t, new_local);
153 }
154
155 mark_scene_dirty();
156 }

158 void editable_world_impl::add_component_by_hash(entity e, size_t component_hash) {
159 const auto& registry = component_operation_registry::get_registry();
160 auto it = registry.find(component_hash);
161
162 if (it == registry.end()) {
163 std::cerr << "[Vesta EditableWorld] Bilesen operasyonu bulunamadi: " << component_hash << '\n';
164 return;
165 }
166
167 if (it->second.add_fn) {
168 it->second.add_fn(e, m_registry);
169 mark_scene_dirty();
170 }
171 }

173 void editable_world_impl::remove_component_by_hash(entity e, size_t component_hash) {
174 // Dış sistemlerin (Fizik, Ses) bileşen yok edilmeden önce haberdar olması için olay dağıtımı yap
175 editor_events::dispatch_on_remove(component_hash, e, m_registry);
176
177 const auto& registry = component_operation_registry::get_registry();
178 auto it = registry.find(component_hash);
179
180 if (it == registry.end()) {
181 return;
182 }
183
184 if (it->second.remove_fn) {
185 it->second.remove_fn(e, m_registry);
186 mark_scene_dirty();
187 }
188 }

253 std::optional<std::string> editable_world_impl::get_asset_type(const std::string& path) const {
254 namespace fs = std::filesystem;
255
256 if (!fs::exists(path)) {
257 return std::nullopt;
258 }
259
260 const std::string ext = fs::path(path).extension().string();
261
262 if (ext == ".mat" || ext == ".json" || ext == ".amat") {
263 return "material";
264 }
265
266 if (ext == ".mesh") {
267 return "mesh";
268 }
269
270 if (ext == ".png" || ext == ".jpg" || ext == ".jpeg" || ext == ".tga" || ext == ".dds") {
271 return "texture";
272 }
273
274 if (ext == ".prefab") {
275 return "prefab";
276 }
277
278 if (ext == ".scene") {
279 return "scene";
280 }
281
282 if (ext == ".shader") {
283 return "shader";
284 }
285
286 return std::nullopt;
287 }
