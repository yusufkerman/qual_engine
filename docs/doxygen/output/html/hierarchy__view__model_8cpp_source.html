14 void hierarchy_view_model::gather_children_recursive(entity parent, std::vector<entity>& out_children) const {
15 if (auto* hc = m_registry.get_component<hierarchy_component>(parent)) {
16 entity current_child = hc->first_child;
17
18 // Kardeş (sibling) zincirini dolaşarak tüm doğrudan çocukları topla
19 while (current_child != 0) {
20 out_children.push_back(current_child);
21 if (auto* child_hc = m_registry.get_component<hierarchy_component>(current_child)) {
22 current_child = child_hc->next_sibling;
23 }
24 else {
25 break;
26 }
27 }
28 }
29 }

31 hierarchy_node_data hierarchy_view_model::get_node_data(entity e) const {
32 hierarchy_node_data data;
33 data.id = e;
34 data.is_selected = m_selection_manager.is_entity_selected(e);
35
36 if (auto* name_comp = m_registry.get_component<name_component>(e)) {
37 data.name = name_comp->name.empty() ? "Unnamed Entity" : name_comp->name;
38 }
39 else {
40 data.name = "Entity " + std::to_string(e);
41 }
42
43 if (auto* hc = m_registry.get_component<hierarchy_component>(e)) {
44 data.parent_id = hc->parent;
45 data.has_children = (hc->first_child != 0);
46 gather_children_recursive(e, data.children);
47 }
48
49 return data;
50 }

52 std::vector<hierarchy_node_data> hierarchy_view_model::get_root_nodes() const {
53 std::vector<hierarchy_node_data> root_nodes;
54 const auto& all_entities = m_registry.get_all_entities();
55
56 for (entity e : all_entities) {
57 bool is_root = true;
58 if (auto* hc = m_registry.get_component<hierarchy_component>(e)) {
59 if (hc->parent != 0) {
60 is_root = false;
61 }
62 }
63
64 if (is_root) {
65 root_nodes.push_back(get_node_data(e));
66 }
67 }
68 return root_nodes;
69 }
