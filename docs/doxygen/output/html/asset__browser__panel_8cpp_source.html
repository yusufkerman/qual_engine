24 void asset_browser_panel::render() {
25 ImGui::Begin("Asset Browser");
26
27 // Yenileme (Refresh) Butonu
28 if (ImGui::Button("Refresh")) {
29 m_vm.refresh();
30 }
31
32 ImGui::Separator();
33
34 // Boş alana sol tıklanınca mevcut seçimi temizle
35 if (ImGui::IsMouseClicked(ImGuiMouseButton_Left) && ImGui::IsWindowHovered(ImGuiHoveredFlags_AllowWhenBlockedByPopup)) {
36 if (!ImGui::IsAnyItemHovered()) {
37 m_vm.clear_selection();
38 }
39 }
40
41 // ========================================================================
42 // BOŞ ALANA SAĞ TIK MENÜSÜ (KÖK DİZİNE VARLIK EKLEME)
43 // ========================================================================
44 if (ImGui::BeginPopupContextWindow("AssetBrowserContextMenu", ImGuiPopupFlags_MouseButtonRight | ImGuiPopupFlags_NoOpenOverItems)) {
45 if (ImGui::MenuItem("➕ Create Material...")) {
46 m_context_target_path = "Assets";
47 m_open_create_popup = true;
48 snprintf(m_create_buffer, sizeof(m_create_buffer), "NewMaterial");
49 }
50 ImGui::EndPopup();
51 }
52
53 // ========================================================================
54 // KÖK (ROOT) DÜĞÜMÜN ÇİZİMİ
55 // ========================================================================
56 const auto& root = m_vm.root();
57 ImGuiTreeNodeFlags root_flags = ImGuiTreeNodeFlags_OpenOnArrow | ImGuiTreeNodeFlags_DefaultOpen | ImGuiTreeNodeFlags_SpanAvailWidth;
58
59 if (m_vm.is_selected("Assets")) {
60 root_flags |= ImGuiTreeNodeFlags_Selected;
61 }
62
63 bool root_open = ImGui::TreeNodeEx("📁 Assets", root_flags);
64 if (ImGui::IsItemClicked()) {
65 m_vm.select("Assets");
66 }
67
68 if (root_open) {
69 for (const auto& child : root.children) {
70 draw_node(child);
71 }
72 ImGui::TreePop();
73 }
74
75 // ========================================================================
76 // MODAL POPUP: YENİ MATERYAL OLUŞTURMA
77 // ========================================================================
78 if (m_open_create_popup) {
79 ImGui::OpenPopup("Create Asset Name");
80 m_open_create_popup = false;
81 }
82
83 if (ImGui::BeginPopupModal("Create Asset Name", NULL, ImGuiWindowFlags_AlwaysAutoResize)) {
84 ImGui::Text("Enter material name:");
85 ImGui::InputText("##create_name", m_create_buffer, sizeof(m_create_buffer));
86 ImGui::Spacing();
87
88 if (ImGui::Button("Create", ImVec2(120, 0))) {
89 m_vm.request_create_material(m_context_target_path, m_create_buffer);
90 ImGui::CloseCurrentPopup();
91 }
92
93 ImGui::SameLine();
94
95 if (ImGui::Button("Cancel", ImVec2(120, 0))) {
96 ImGui::CloseCurrentPopup();
97 }
98
99 ImGui::EndPopup();
100 }
101
102 // ========================================================================
103 // MODAL POPUP: YENİDEN ADLANDIRMA (RENAME)
104 // ========================================================================
105 if (m_open_rename_popup) {
106 ImGui::OpenPopup("Rename Asset");
107 m_open_rename_popup = false;
108 }
109
110 if (ImGui::BeginPopupModal("Rename Asset", NULL, ImGuiWindowFlags_AlwaysAutoResize)) {
111 ImGui::Text("Rename '%s' to:", m_context_target_path.filename().string().c_str());
112 ImGui::InputText("##rename_input", m_rename_buffer, sizeof(m_rename_buffer));
113 ImGui::Spacing();
114
115 if (ImGui::Button("OK", ImVec2(120, 0))) {
116 m_vm.request_rename(m_context_target_path, m_rename_buffer);
117 ImGui::CloseCurrentPopup();
118 }
119
120 ImGui::SameLine();
121
122 if (ImGui::Button("Cancel", ImVec2(120, 0))) {
123 ImGui::CloseCurrentPopup();
124 }
125
126 ImGui::EndPopup();
127 }
128
129 ImGui::End();
130 }

132 void asset_browser_panel::draw_node(const asset_file_entry& entry) {
133 std::string path_str = entry.full_path.string();
134
135 // Aynı isimde klasörler olabileceği için ImGui state karışmasını engelliyoruz
136 ImGui::PushID(path_str.c_str());
137
138 std::string prefix = (entry.type == asset_type::folder) ? "📁 " : "📄 ";
139 std::string label = prefix + entry.name;
140 bool is_selected = m_vm.is_selected(path_str);
141
142 ImGuiTreeNodeFlags flags = ImGuiTreeNodeFlags_OpenOnArrow | ImGuiTreeNodeFlags_SpanAvailWidth;
143 if (entry.type != asset_type::folder) {
144 flags |= ImGuiTreeNodeFlags_Leaf;
145 }
146 if (is_selected) {
147 flags |= ImGuiTreeNodeFlags_Selected;
148 }
149
150 bool is_open = ImGui::TreeNodeEx(label.c_str(), flags);
151
152 // ========================================================================
153 // FARE ETKİLEŞİMLERİ (ÇİFT TIKLAMA VE SAĞ TIK SEÇİMİ)
154 // ========================================================================
155 if (ImGui::IsItemHovered()) {
156 if (ImGui::IsMouseDoubleClicked(ImGuiMouseButton_Left)) {
157 m_vm.select_and_notify(path_str);
158 }
159
160 if (ImGui::IsMouseClicked(ImGuiMouseButton_Right)) {
161 m_vm.select(path_str);
162 }
163 }
164
165 // ========================================================================
166 // ÖĞEYE ÖZEL İÇERİK MENÜSÜ (CONTEXT MENU)
167 // ========================================================================
168 if (ImGui::BeginPopupContextItem("ItemContextMenu")) {
169 m_vm.select_and_notify(path_str);
170
171 if (entry.type == asset_type::folder) {
172 if (ImGui::MenuItem("➕ Create Material Here...")) {
173 m_context_target_path = entry.full_path;
174 m_open_create_popup = true;
175 snprintf(m_create_buffer, sizeof(m_create_buffer), "NewMaterial");
176 }
177 ImGui::Separator();
178 }
179
180 if (ImGui::MenuItem("📝 Rename")) {
181 m_context_target_path = entry.full_path;
182 snprintf(m_rename_buffer, sizeof(m_rename_buffer), "%s", entry.full_path.stem().string().c_str());
183 m_open_rename_popup = true;
184 }
185
186 if (ImGui::MenuItem("👯 Duplicate / Copy")) {
187 m_vm.request_duplicate(entry.full_path);
188 }
189
190 ImGui::Separator();
191
192 if (ImGui::MenuItem("🗑️ Delete")) {
193 m_vm.request_delete(entry.full_path);
194 }
195
196 ImGui::EndPopup();
197 }
198
199 // ========================================================================
200 // SÜRÜKLE VE BIRAK (DRAG & DROP) KAYNAĞI
201 // ========================================================================
202 if (ImGui::BeginDragDropSource(ImGuiDragDropFlags_SourceAllowNullID))
203 {
204 vesta::asset_drag_payload payload{};
205 std::string ext = entry.full_path.extension().string();
206 std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
207
208 if (ext == ".json" || ext == ".amat") {
209 payload.type = vesta::asset_payload_type::material;
210 }
211 else if (ext == ".png" || ext == ".jpg" || ext == ".tga" || ext == ".dds") {
212 payload.type = vesta::asset_payload_type::texture;
213 }
214 else if (ext == ".obj" || ext == ".fbx" || ext == ".gltf" || ext == ".dae") {
215 payload.type = vesta::asset_payload_type::model;
216 }
217 else {
218 payload.type = vesta::asset_payload_type::model; // Fallback
219 }
220
221 // Dosya yolunu bellekte kopyalanabilir (POD) bir şekilde pakete ekle
222 snprintf(payload.path, sizeof(payload.path), "%s", path_str.c_str());
223 payload.guid = entry.guid;
224
225 ImGui::SetDragDropPayload("ARIA_ASSET_PAYLOAD", &payload, sizeof(payload));
226
227 // Sürüklerken farenin altında görünecek olan önizleme metni
228 ImGui::Text("%s %s", (entry.type == asset_type::folder ? "📁" : "📄"), entry.name.c_str());
229 ImGui::EndDragDropSource();
230 }
231
232 // ========================================================================
233 // SÜRÜKLE VE BIRAK (DRAG & DROP) HEDEFİ
234 // ========================================================================
235 if (entry.type == asset_type::folder) {
236 if (ImGui::BeginDragDropTarget())
237 {
238 if (const ImGuiPayload* imgui_payload = ImGui::AcceptDragDropPayload("ARIA_ASSET_PAYLOAD"))
239 {
240 auto* data = (const vesta::asset_drag_payload*)imgui_payload->Data;
241 std::filesystem::path src(data->path);
242
243 // ViewModel üzerinden dosya taşıma (move) isteği yolla
244 m_vm.request_move(src, entry.full_path);
245 }
246 ImGui::EndDragDropTarget();
247 }
248 }
249
250 if (is_open) {
251 if (entry.type == asset_type::folder) {
252 for (const auto& child : entry.children) {
253 draw_node(child);
254 }
255 }
256 ImGui::TreePop();
257 }
258
259 ImGui::PopID();
260 }
