81void skybox_pass::render(EngineContext& ctx, const glm::mat4& view, const glm::mat4& projection)
82{
83 if (!ctx.aria_system || ctx.skyboxTexture == 0) {
84 return;
85 }
86
87 const unsigned int skyboxGLShaderID = ctx.aria_system->get_shader_id(ctx.skyboxShaderHandle);
88 if (skyboxGLShaderID == 0) {
89 return;
90 }
91
92 if (s_skyVAO == 0) {
93 initSkybox();
94 }
95
96 // Skybox en uzakta çizileceği için derinlik testini LEQUAL yap ve derinlik yazımını kapat
97 glDepthFunc(GL_LEQUAL);
98 glDepthMask(GL_FALSE);
99
100 glUseProgram(skyboxGLShaderID);
101
102 // Kameranın dünya konumunu (ötelemesini) sıfırla, yalnızca dönüşü koru
103 const glm::mat4 skyView = glm::mat4(glm::mat3(view));
104
105 const GLint skyboxLoc = glGetUniformLocation(skyboxGLShaderID, "skybox");
106 if (skyboxLoc != -1) {
107 glUniform1i(skyboxLoc, 0);
108 }
109
110 const GLint viewLoc = glGetUniformLocation(skyboxGLShaderID, "view");
111 if (viewLoc != -1) {
112 glUniformMatrix4fv(viewLoc, 1, GL_FALSE, glm::value_ptr(skyView));
113 }
114
115 const GLint projLoc = glGetUniformLocation(skyboxGLShaderID, "projection");
116 if (projLoc != -1) {
117 glUniformMatrix4fv(projLoc, 1, GL_FALSE, glm::value_ptr(projection));
118 }
119
120 // Küp haritası dokusunu bağla ve geometriyi çiz
121 glBindVertexArray(s_skyVAO);
122 glActiveTexture(GL_TEXTURE0);
123 glBindTexture(GL_TEXTURE_CUBE_MAP, ctx.skyboxTexture);
124
125 glDrawArrays(GL_TRIANGLES, 0, 36);
126
127 // OpenGL durumlarını varsayılana geri yükle
128 glBindVertexArray(0);
129 glUseProgram(0);
130 glDepthMask(GL_TRUE);
131 glDepthFunc(GL_LESS);
132}
