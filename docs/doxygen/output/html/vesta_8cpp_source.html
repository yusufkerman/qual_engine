29 void vesta_system::init(aria::system* ariaSys, EngineContext& engineCtx, EditorContext& editorCtx)
30 {
31 std::cout << "[Vesta] Editor alt sistemleri baslatiliyor...\n";
32
33 // Güvenli yeniden başlatma (re-init) için panelleri temizle
34 m_panels.clear();
35
36 // 1. Bileşen (Component) Arayüz Kayıtları
37 register_transform_ui_system();
38 register_camera_ui_system();
39 register_light_ui_system();
40 register_rigidbody_ui_system();
41 register_collider_ui_system();
42 register_native_script_ui_system();
43 register_material_ui_system(*engineCtx.aria_system);
44 register_mesh_ui_system(*engineCtx.aria_system);
45
46 // 2. Çekirdek Servislerin Hazırlanması
47 registry& reg = engineCtx.session.game_scene().get_registry();
48
49 m_editable_world = std::make_unique<editable_world_impl>(reg, ariaSys);
50 m_command_stack = std::make_unique<command_stack>(*m_editable_world);
51 m_material_service = std::make_unique<material_service>(*m_editable_world, *m_command_stack);
52 m_selection_manager = std::make_unique<editor_selection_manager>();
53
54 // 3. Varlık (Asset) Arayüz Kayıtları
55 register_material_asset_ui_system(engineCtx.aria_system, m_material_service.get());
56
57 // 4. Editör Panelleri ve ViewModel'lerin Başlatılması
58 init_top_bar(engineCtx, editorCtx);
59 init_scene_viewport(engineCtx);
60 init_game_viewport(engineCtx);
61 init_hierarchy(reg);
62 init_inspector(reg);
63 init_asset_browser(*ariaSys);
64
65 // 5. Fizik Kancalarının (Hooks) Uygulanması
66 auto& physApi = engineCtx.physicsSystem.api();
67 apply_hooks(physApi);
68 }

77 void vesta_system::apply_hooks(physics_api& api)
78 {
79 // Rigidbody silindiğinde fizik dünyasından da temizlenmesi için kanca eklendi
80 vesta::vesta_module::register_remove_hook<rigidbody_component>(
81 [&api](entity e)
82 {
83 api.destroy_body(e);
84 });
85
86 // Collider silindiğinde fizik dünyasından şeklinin kaldırılması için kanca eklendi
87 vesta::vesta_module::register_remove_hook<collider_component>(
88 [&api](entity e)
89 {
90 api.remove_collider_shape(e);
91 });
92 }

94 void vesta_system::init_top_bar(EngineContext& engineCtx, EditorContext& editorCtx)
95 {
96 m_top_bar_vm = std::make_unique<top_bar_view_model>(
97 editorCtx,
98 [&engineCtx](const std::string& path)
99 {
100 if (engineCtx.sera)
101 {
102 engineCtx.sera->save_scene(path, engineCtx.session.editor_scene());
103 }
104 },
105 [&engineCtx](const std::string& path)
106 {
107 if (engineCtx.sera)
108 {
109 engineCtx.sera->load_scene(path, engineCtx.session.editor_scene());
110 }
111 }
112 );
113
114 m_top_bar_panel = std::make_unique<top_bar_panel>(*m_top_bar_vm);
115 register_panel(m_top_bar_panel.get());
116 }

165 void vesta_system::init_asset_browser(aria::system& ariaSys)
166 {
167 m_asset_browser_system = std::make_unique<asset_browser_system>(
168 "Assets",
169 &ariaSys,
170 m_material_service.get()
171 );
172
173 m_asset_browser_system->refresh();
174
175 m_asset_browser_vm = std::make_unique<asset_browser_view_model>(
176 *m_asset_browser_system,
177 *m_material_service,
178 [this](const std::string& path)
179 {
180 if (path.empty()) {
181 m_selection_manager->clear_asset_selection();
182 }
183 else {
184 m_selection_manager->select_asset(path);
185 }
186 },
187 [this]()
188 {
189 m_selection_manager->clear_entity_selection();
190 }
191 );
192
193 m_asset_browser_panel = std::make_unique<asset_browser_panel>(*m_asset_browser_vm);
194 m_asset_browser_panel->init();
195
196 register_panel(m_asset_browser_panel.get());
197 }
