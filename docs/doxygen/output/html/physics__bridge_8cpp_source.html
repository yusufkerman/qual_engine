21 void create_physics_world(EngineContext& ctx, scene& current_scene)
22 {
23 auto* jolt_system = ctx.physicsSystem.get_jolt_system();
24 auto& body_interface = jolt_system->GetBodyInterface();
25 auto& reg = current_scene.get_registry();
26
27 auto& rb_pool = reg.get_all<rigidbody_component>();
28
29 for (auto& [ent, rb] : rb_pool)
30 {
31 auto* transform = reg.get_component<transform_component>(ent);
32 auto* collider = reg.get_component<collider_component>(ent);
33
34 if (!transform || !collider)
35 {
36 continue;
37 }
38
39 JPH::ShapeRefC shape;
40 if (collider->type == collider_type::box)
41 {
42 shape = new JPH::BoxShape(JPH::Vec3(collider->extents.x, collider->extents.y, collider->extents.z));
43 }
44 else if (collider->type == collider_type::sphere)
45 {
46 shape = new JPH::SphereShape(collider->extents.x);
47 }
48
49 JPH::EMotionType jolt_motion = JPH::EMotionType::Static;
50 JPH::ObjectLayer jolt_layer = layers::NON_MOVING;
51
52 if (rb.motion == motion_type::dynamic_body)
53 {
54 jolt_motion = JPH::EMotionType::Dynamic;
55 jolt_layer = layers::MOVING;
56 }
57 else if (rb.motion == motion_type::kinematic_body)
58 {
59 jolt_motion = JPH::EMotionType::Kinematic;
60 jolt_layer = layers::MOVING;
61 }
62
63 // Euler rotasyon değerlerinden dönüşüm matrisi üzerinden güvenli kuaterniyon eldesi (Sıra: X -> Y -> Z)
64 glm::mat4 rotMatrix = glm::mat4(1.0f);
65 rotMatrix = glm::rotate(rotMatrix, transform->rotation.x, glm::vec3(1.0f, 0.0f, 0.0f));
66 rotMatrix = glm::rotate(rotMatrix, transform->rotation.y, glm::vec3(0.0f, 1.0f, 0.0f));
67 rotMatrix = glm::rotate(rotMatrix, transform->rotation.z, glm::vec3(0.0f, 0.0f, 1.0f));
68
69 const glm::quat q = glm::quat_cast(rotMatrix);
70 const JPH::Quat jolt_quat(q.x, q.y, q.z, q.w);
71
72 JPH::BodyCreationSettings body_settings(
73 shape,
74 JPH::RVec3(transform->position.x, transform->position.y, transform->position.z),
75 jolt_quat,
76 jolt_motion,
77 jolt_layer
78 );
79
80 body_settings.mFriction = rb.friction;
81 body_settings.mRestitution = rb.restitution;
82
83 if (rb.motion == motion_type::dynamic_body)
84 {
85 body_settings.mOverrideMassProperties = JPH::EOverrideMassProperties::CalculateInertia;
86 body_settings.mMassPropertiesOverride.mMass = rb.mass;
87 }
88
89 JPH::EAllowedDOFs allowedDOFs = JPH::EAllowedDOFs::All;
90 if (rb.freeze_rotation_x) allowedDOFs &= ~JPH::EAllowedDOFs::RotationX;
91 if (rb.freeze_rotation_y) allowedDOFs &= ~JPH::EAllowedDOFs::RotationY;
92 if (rb.freeze_rotation_z) allowedDOFs &= ~JPH::EAllowedDOFs::RotationZ;
93
94 body_settings.mAllowedDOFs = allowedDOFs;
95
96 JPH::Body* body = body_interface.CreateBody(body_settings);
97 body->SetUserData(static_cast<JPH::uint64>(ent));
98 body_interface.AddBody(body->GetID(), JPH::EActivation::Activate);
99
100 rb.body_id = body->GetID();
101 }
102
103 jolt_system->OptimizeBroadPhase();
104 }

106 void sync_physics_to_ecs(EngineContext& ctx, scene& current_scene)
107 {
108 auto* jolt_system = ctx.physicsSystem.get_jolt_system();
109 auto& body_interface = jolt_system->GetBodyInterface();
110 auto& reg = current_scene.get_registry();
111
112 auto& rb_pool = reg.get_all<rigidbody_component>();
113
114 for (auto& [ent, rb] : rb_pool)
115 {
116 if (rb.body_id.IsInvalid()) continue;
117 if (rb.motion == motion_type::static_body) continue;
118
119 auto* transform = reg.get_component<transform_component>(ent);
120 if (!transform) continue;
121
122 // Konum eşitleme
123 const JPH::RVec3 post_sim_pos = body_interface.GetPosition(rb.body_id);
124 transform->position = glm::vec3(post_sim_pos.GetX(), post_sim_pos.GetY(), post_sim_pos.GetZ());
125
126 // Rotasyon eşitleme (Render altyapısı ile uyumluluk için güvenli Euler çıkarımı)
127 if (rb.motion == motion_type::dynamic_body)
128 {
129 const JPH::Quat post_sim_quat = body_interface.GetRotation(rb.body_id);
130
131 // GLM Kuaterniyon kurucusu sıralaması: w, x, y, z
132 const glm::quat q(post_sim_quat.GetW(), post_sim_quat.GetX(), post_sim_quat.GetY(), post_sim_quat.GetZ());
133 const glm::mat4 rotMatrix = glm::mat4_cast(q);
134
135 float rotX, rotY, rotZ;
136 // Motorun genel X -> Y -> Z matris kurgusu ile uyumlu ters(çözüm) işlemidir
137 glm::extractEulerAngleXYZ(rotMatrix, rotX, rotY, rotZ);
138
139 transform->rotation = glm::vec3(rotX, rotY, rotZ);
140 }
141 }
142 }

144 void sync_ecs_to_physics(EngineContext& ctx, scene& current_scene)
145 {
146 auto* jolt_system = ctx.physicsSystem.get_jolt_system();
147 auto& body_interface = jolt_system->GetBodyInterface();
148 auto& reg = current_scene.get_registry();
149
150 auto& rb_pool = reg.get_all<rigidbody_component>();
151 bool any_static_or_kinematic_moved = false;
152
153 for (auto& [ent, rb] : rb_pool)
154 {
155 if (rb.body_id.IsInvalid()) continue;
156 if (rb.motion == motion_type::kinematic_body) continue;
157
158 auto* transform = reg.get_component<transform_component>(ent);
159 if (!transform) continue;
160
161 const JPH::RVec3 jolt_pos = body_interface.GetPosition(rb.body_id);
162 const glm::vec3 ecs_pos = transform->position;
163
164 if (glm::distance(glm::vec3(jolt_pos.GetX(), jolt_pos.GetY(), jolt_pos.GetZ()), ecs_pos) > 0.005f)
165 {
166 glm::mat4 rotMatrix = glm::mat4(1.0f);
167 rotMatrix = glm::rotate(rotMatrix, transform->rotation.x, glm::vec3(1.0f, 0.0f, 0.0f));
168 rotMatrix = glm::rotate(rotMatrix, transform->rotation.y, glm::vec3(0.0f, 1.0f, 0.0f));
169 rotMatrix = glm::rotate(rotMatrix, transform->rotation.z, glm::vec3(0.0f, 0.0f, 1.0f));
170
171 const glm::quat q = glm::quat_cast(rotMatrix);
172 const JPH::Quat jolt_quat(q.x, q.y, q.z, q.w);
173
174 body_interface.SetPositionAndRotation(
175 rb.body_id,
176 JPH::RVec3(ecs_pos.x, ecs_pos.y, ecs_pos.z),
177 jolt_quat,
178 JPH::EActivation::Activate
179 );
180
181 if (rb.motion == motion_type::static_body || rb.motion == motion_type::kinematic_body)
182 {
183 any_static_or_kinematic_moved = true;
184 }
185 }
186 }
187
188 if (any_static_or_kinematic_moved)
189 {
190 jolt_system->OptimizeBroadPhase();
191
192 for (auto& [ent, rb] : rb_pool)
193 {
194 if (rb.body_id.IsInvalid()) continue;
195 if (rb.motion == motion_type::dynamic_body)
196 {
197 body_interface.ActivateBody(rb.body_id);
198 }
199 }
200 }

203 void destroy_physics_world(EngineContext& ctx, scene& current_scene)
204 {
205 auto* jolt_system = ctx.physicsSystem.get_jolt_system();
206 auto& body_interface = jolt_system->GetBodyInterface();
207 auto& reg = current_scene.get_registry();
208

210

213 if (rb.body_id.IsInvalid()) continue;
214
