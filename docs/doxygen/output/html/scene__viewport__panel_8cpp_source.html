19 void scene_viewport_panel::render()
20 {
21 ImGui::Begin(m_title.c_str(), &m_is_open);
22
23 auto& scene_state = m_viewport_state.scene();
24
25 // Girdi (Input), kısayol ve fly-cam kontrolleri için odak/imleç durumlarını güncelle
26 scene_state.focused = ImGui::IsWindowFocused(ImGuiFocusedFlags_RootAndChildWindows);
27 scene_state.hovered = ImGui::IsWindowHovered(ImGuiFocusedFlags_RootAndChildWindows);
28
29 ImGui::PushID("SceneViewport");
30
31 // Render alanının pencereye tam oturması için kenar boşluklarını sıfırla
32 ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(0, 0));
33 ImGui::PushStyleVar(ImGuiStyleVar_WindowBorderSize, 0.0f);
34
35 ImVec2 avail = ImGui::GetContentRegionAvail();
36 if (avail.x > 1.0f && avail.y > 1.0f)
37 {
38 scene_state.size = { avail.x, avail.y };
39 }
40
41 unsigned int tex = m_vm.get_texture_id();
42 if (avail.x > 0.0f && avail.y > 0.0f && tex != 0)
43 {
44 ImVec2 cursor = ImGui::GetCursorScreenPos();
45
46 // Gizmo ve raycast/picking işlemleri için sol üst ekran koordinatını kaydet
47 scene_state.position = { cursor.x, cursor.y };
48
49 // Çizimin panel sınırları dışına taşmasını engellemek için kırpma (clipping) uygula
50 ImGui::GetWindowDrawList()->PushClipRect(
51 cursor,
52 ImVec2(cursor.x + avail.x, cursor.y + avail.y),
53 true
54 );
55
56 // OpenGL UV koordinatlarını (0,1 -> 1,0) ters çevirerek FBO dokusunu bas
57 ImGui::Image(
58 (ImTextureID)(intptr_t)tex,
59 avail,
60 ImVec2(0, 1),
61 ImVec2(1, 0)
62 );
63
64 ImGui::GetWindowDrawList()->PopClipRect();
65 }
66
67 ImGui::PopStyleVar(2);
68 ImGui::PopID();
69 ImGui::End();
