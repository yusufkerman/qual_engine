93entity registry::duplicate_entity(entity source)
94{

103 {
104 auto& nameComp = add_component<name_component>(newEntity, *c);
105 nameComp.name += "_Copy";
106 }
107 else
108 {
109 add_component<name_component>(newEntity, name_component{ "Entity_Copy" });
110 }
111
112 return newEntity;
