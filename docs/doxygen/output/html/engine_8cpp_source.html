74 static void handle_mode_transition(EngineContext& ctx, EditorContext& editorCtx)
75 {
76 // Mod değişikliği yoksa işlem yapma
77 if (editorCtx.mode == ctx.prevMode)
78 return;
79
80 // Editör kamerası fare durumunu sıfırla
81 if (editorCtx.camera)
82 {
83 editorCtx.camera->resetMouseState();
84 }
85
86 ctx.runtimeCameraState = fly_camera_state{};
87 glfwSetInputMode(ctx.window, GLFW_CURSOR, GLFW_CURSOR_NORMAL);
88
89 if (editorCtx.mode == EngineMode::PLAY)
90 {
91 ctx.session.play(ctx);
92 physics_bridge::create_physics_world(ctx, ctx.session.game_scene());
93 }
94 else
95 {
96 physics_bridge::destroy_physics_world(ctx, ctx.session.runtime_scene());
97 ctx.session.stop();
98 }
99
100 ctx.prevMode = editorCtx.mode;
101 }

104bool Engine::init(EngineContext& ctx, EditorContext& editorCtx)
105{
106 if (!glfwInit())
107 return false;
108
109 glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 3);
110 glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 3);
111 glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
112
113 ctx.window = glfwCreateWindow(ctx.screenW, ctx.screenH, "PS1 Engine", nullptr, nullptr);
114 if (!ctx.window)
115 return false;
116
117 glfwMakeContextCurrent(ctx.window);
118
119 // OpenGL fonksiyonlarını (GLAD) yükle
120 if (!gladLoadGLLoader((GLADloadproc)glfwGetProcAddress))
121 return false;
122
123 // Aria Varlık (Asset) Sistemini başlat (Not: GLAD'den hemen sonra olmalıdır)
124 if (!ctx.aria_system) {
125 ctx.aria_system = new aria::system();
126 }
127 ctx.aria_system->init(); // Varsayılan shader ve dokuları belleğe yükler
128
129 // Sera Sistemini başlat
130 ctx.sera = new sera::sera();
131 ctx.sera->init(ctx);
132
133 // Gölge tamponunu (Shadow FBO) başlat
134 if (!ctx.shadowFBO.init(2048, 2048))
135 {
136 std::cout << "[Engine] Hata: Gölge sistemi başlatılamadı!" << std::endl;
137 }
138
139 ctx.renderSystem.init();
140
141 glEnable(GL_DEPTH_TEST);
142 glDisable(GL_CULL_FACE);
143
144 // GLFW kullanıcı veri işaretçisini (User Pointer) ayarla
145 static WindowUserData userData;
146 userData.engineCtx = &ctx;
147 userData.editorCtx = &editorCtx;
148
149 glfwSetWindowUserPointer(ctx.window, &userData);
150 glfwSetFramebufferSizeCallback(ctx.window, framebuffer_size_callback);
151 glfwSetScrollCallback(ctx.window, scroll_callback);
152
153 // Arayüz ve görsel sistemleri başlat
154 ctx.uiManager.init(ctx.window);
155 ctx.grid.init();
156 ctx.gizmoSystem.init();
157
158 // Fizik sistemini başlat
159 if (!ctx.physicsSystem.init(ctx.session.game_scene()))
160 {
161 std::cout << "[Physics] Hata: init basarisiz.\n";
162 return false;
163 }
164
165 // Gerekli Shader'ları Aria üzerinden içe aktar
166 ctx.defaultShaderHandle = asset_pipeline::import_shader(ctx, "Shaders/default_shader");
167 ctx.gizmoShaderHandle = asset_pipeline::import_shader(ctx, "Shaders/Debug/gizmo");
168 ctx.shadowShaderHandle = asset_pipeline::import_shader(ctx, "Shaders/shadow");
169 ctx.gridShaderHandle = asset_pipeline::import_shader(ctx, "Shaders/Debug/grid");
170 ctx.skyboxShaderHandle = asset_pipeline::import_shader(ctx, "Shaders/Skybox/skybox");
171
172 // Ana shader için ilk uniform değerlerini bağla
173 unsigned int mainGLid = ctx.aria_system->get_shader_id(ctx.defaultShaderHandle);
174 if (mainGLid != 0)
175 {
176 glUseProgram(mainGLid);
177 glUniform1i(glGetUniformLocation(mainGLid, "diffuseTex"), 0);
178 glUniform1i(glGetUniformLocation(mainGLid, "normalTex"), 1);
179 glUseProgram(0);
180 }
181
182 ctx.editorTarget.create(1, 1);
183 ctx.gameTarget.create(1, 1);
184
185 ctx.vestaSys.init(ctx.aria_system, ctx, editorCtx);
186
187 // Editör modundaysa ve kamera henüz oluşturulmadıysa başlat
188 if (ctx.isEditorBuild && !editorCtx.camera)
189 {
190 editorCtx.camera = new EditorCamera();
191 }
192
193 init_input(ctx);
194 ctx.prevMode = editorCtx.mode;
195
196 // Gökyüzü (Skybox) küp haritasını yükle
197 ctx.skyboxTexture = cubemap_loader::load_atlas("Assets/sky/skybox1.png");
198
199 if (NFD_Init() != NFD_OKAY)
200 {
201 std::cout << "[NFD] Hata: init basarisiz.\n";
202 }
203
204 std::cout << "[Engine] Basariyla baslatildi. Default handle index="
205 << ctx.defaultShaderHandle.index
206 << " gen=" << ctx.defaultShaderHandle.generation << "\n";
207
208 return true;
209}

211void Engine::update(EngineContext& ctx, EditorContext& editorCtx)
212{
213 gameplay::context::set_engine_context(&ctx);
214 gameplay::context::set_active_scene(&ctx.session.game_scene());
215
216 ctx.inputSystem.update(ctx, editorCtx);
217 handle_mode_transition(ctx, editorCtx);
218
219 runtime_layer::update(ctx, editorCtx);
220
221 if (editorCtx.mode == EngineMode::PLAY)
222 {
223 // TODO: Sabit delta time (dt) ileride dinamik zaman ölçümüne çevrilmelidir.
224 float dt = 0.016f;
225 ctx.inputState.active = (!ctx.isEditorBuild) || (ctx.vestaSys.viewport_state().game().focused);
226 ctx.session.game_scene().on_update_runtime(dt);
227 }
228
229 ctx.session.game_scene().update_light_batch();
230 ctx.cameraSystem.update(ctx.session.game_scene());
231
232 if (ctx.isEditorBuild)
233 {
234 editor_layer::update(ctx, editorCtx);
235 ctx.session.editor_scene().update_light_batch();
236 }
237}

254void Engine::shutdown(EngineContext& ctx, EditorContext& editorCtx)
255{
256 ctx.session.game_scene().clean_up_scripts();
257 ctx.uiManager.shutdown();
258
259 ctx.editorTarget.destroy();
260 ctx.gameTarget.destroy();
261 ctx.grid.shutdown();
262 ctx.physicsSystem.shutdown();
263
264 if (editorCtx.camera)
265 {
266 delete editorCtx.camera;
267 editorCtx.camera = nullptr;
268 }
269
270 // Sera sistemini temizle
271 if (ctx.sera)
272 {
273 delete ctx.sera;
274 ctx.sera = nullptr;
275 }
276
277 // Aria'nın tüm GPU ve bellek kaynaklarını serbest bırak
278 if (ctx.aria_system) {
279 ctx.aria_system->clear_all_resources();
280 ctx.aria_system->shutdown();
281 delete ctx.aria_system;
282 ctx.aria_system = nullptr;
283 }
284
285 if (ctx.window)
286 {
287 glfwDestroyWindow(ctx.window);
288 ctx.window = nullptr;
289 }
290
291 NFD_Quit();
292 glfwTerminate();
293}
