37shader::shader(const char* vertexPath, const char* fragmentPath)
38{
39 const std::string vCode = loadFile(vertexPath);
40 const std::string fCode = loadFile(fragmentPath);
41
42 const char* vShaderCode = vCode.c_str();
43 const char* fShaderCode = fCode.c_str();
44
45 unsigned int vs = glCreateShader(GL_VERTEX_SHADER);
46 glShaderSource(vs, 1, &vShaderCode, NULL);
47 glCompileShader(vs);
48
49 int success;
50 char infoLog[512];
51 glGetShaderiv(vs, GL_COMPILE_STATUS, &success);
52 if (!success)
53 {
54 glGetShaderInfoLog(vs, 512, NULL, infoLog);
55 std::cerr << "[Shader] VERTEX SHADER DERLEME HATASI:\n" << infoLog << '\n';
56 }
57
58 unsigned int fs = glCreateShader(GL_FRAGMENT_SHADER);
59 glShaderSource(fs, 1, &fShaderCode, NULL);
60 glCompileShader(fs);
61
62 glGetShaderiv(fs, GL_COMPILE_STATUS, &success);
63 if (!success)
64 {
65 glGetShaderInfoLog(fs, 512, NULL, infoLog);
66 std::cerr << "[Shader] FRAGMENT SHADER DERLEME HATASI:\n" << infoLog << '\n';
67 }
68
69 ID = glCreateProgram();
70 glAttachShader(ID, vs);
71 glAttachShader(ID, fs);
72 glLinkProgram(ID);
73
74 glGetProgramiv(ID, GL_LINK_STATUS, &success);
75 if (!success)
76 {
77 glGetProgramInfoLog(ID, 512, NULL, infoLog);
78 std::cerr << "[Shader] SHADER BAGLAMA (LINK) HATASI:\n" << infoLog << '\n';
79 }
80
81 glDeleteShader(vs);
82 glDeleteShader(fs);
83}
