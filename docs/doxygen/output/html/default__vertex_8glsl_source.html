1#version 330 core
2
3// ============================================================================
4// GİRDİ (IN) DEĞİŞKENLERİ - VBO (Vertex Buffer Object) Veri Yapısı
5// ============================================================================
6layout(location = 0) in vec3 aPos; // Köşe (Vertex) yerel konumu
7layout(location = 1) in vec3 aNormal; // Köşe yüzey normali
8layout(location = 2) in vec2 aUV; // Doku (Texture) koordinatı
9layout(location = 3) in vec3 aTangent; // Yüzey teğet (Tangent) vektörü
10
11// ============================================================================
12// MATRİS UNIFORM DEĞİŞKENLERİ
13// ============================================================================
14uniform mat4 model; // Yerel uzaydan -> Dünya uzayına dönüşüm matrisi
15uniform mat4 view; // Dünya uzayından -> Kamera (Görünüm) uzayına dönüşüm matrisi
16uniform mat4 projection; // Kamera uzayından -> Ekran (İzdüşüm) uzayına dönüşüm matrisi
17
18// ============================================================================
19// ÇIKTI (OUT) DEĞİŞKENLERİ - Fragment Shader'a aktarılacak veriler
20// ============================================================================
21out vec3 vWorldPos; // Dünya uzayındaki köşe konumu
22out vec3 vNormal; // Dünya uzayındaki normal vektörü
23out vec2 vUV; // Doku koordinatları
24out mat3 vTBN; // Teğet uzayı (Tangent Space) dönüşüm matrisi
25
26// ============================================================================
27// ANA ÇİZİM FONKSİYONU
28// ============================================================================
29void main()
30{
31 // 1. Köşenin dünya uzayındaki konumunun hesaplanması
32 vec4 worldPos = model * vec4(aPos, 1.0);
33 vWorldPos = worldPos.xyz;
34
35 // 2. Normal Matrisi: Eşitsiz ölçeklendirme (Non-uniform scaling) kaynaklı bozulmaları önler
36 mat3 normalMatrix = mat3(transpose(inverse(model)));
37
38 // 3. TBN (Tangent, Bitangent, Normal) Matrisinin Hesaplanması
39 vec3 N = normalize(normalMatrix * aNormal);
40 vec3 T = normalize(normalMatrix * aTangent);
41
42 // Gram-Schmidt Dikleştirme (Orthogonalization):
43 // Teğet vektörünün, normal vektörüne kusursuz bir şekilde dik olmasını garanti altına alır.
44 T = normalize(T - dot(T, N) * N);
45
46 vec3 B = cross(N, T); // Bitangent vektörünün vektörel çarpım ile elde edilmesi
47
48 vTBN = mat3(T, B, N);
49 vNormal = N;
50
51 // 4. Doku koordinatlarının doğrudan aktarılması
52 vUV = aUV;
53
54 // 5. Köşenin son ekran (Clip Space) koordinatının belirlenmesi
55 gl_Position = projection * view * worldPos;
56}
