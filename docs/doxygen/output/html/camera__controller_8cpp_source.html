21void camera_controller::on_update(float dt)
22{
23 // 1. Hedef oyuncu varlığını Gameplay API ile al
24 auto playerEntity = gameplay::scene_service::get_entity(target);
25 if (!playerEntity)
26 {
27 playerEntity = gameplay::scene_service::find_entity("Player");
28 if (!playerEntity)
29 {
30 return;
31 }
32 target = playerEntity.id();
33 }
34
35 // 2. Gameplay Input API üzerinden fare hareket deltasını oku
36 const glm::vec2 mouseDelta = gameplay::input::get_mouse_delta();
37 glm::vec3 rot = transform().get_rotation_euler();
38
39 rot.x -= mouseDelta.y * sensitivity;
40 rot.y += mouseDelta.x * sensitivity;
41
42 // 3. YAW (Sağ-Sol) Radian Wrap (0 ile 2*PI arası)
43 const float twoPi = 2.0f * glm::pi<float>();
44 rot.y = std::fmod(rot.y, twoPi);
45 if (rot.y < 0.0f)
46 {
47 rot.y += twoPi;
48 }
49
50 // 4. PITCH (Yukarı-Aşağı) Clamp (-85 ile +85 derece / 1.48 radyan)
51 constexpr float maxPitchRadian = 1.48f;
52 rot.x = std::clamp(rot.x, -maxPitchRadian, maxPitchRadian);
53
54 // 5. Güncellenen dönüşüm değerlerini transform sarmalayıcısına yaz
55 transform().set_rotation_euler(rot);
56
57 // 6. Kamerayı oyuncunun baş/göz hizasına taşı
58 transform().set_position(playerEntity.transform().get_position() + glm::vec3(0.0f, 3.7f, 0.0f));
59}
