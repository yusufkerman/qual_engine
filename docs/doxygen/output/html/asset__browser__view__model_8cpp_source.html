56 void asset_browser_view_model::request_create_material(
57 const std::filesystem::path& target_dir, const std::string& name)
58 {
59 // Eğer hedeflenen yol bir klasör değilse, bir üst dizinini (parent) baz al
60 std::filesystem::path dir = std::filesystem::is_directory(target_dir)
61 ? target_dir
62 : target_dir.parent_path();
63
64 // İsim çakışmalarını önlemek için sonuna numara ekle (örnek: YeniMateryal_1)
65 std::string unique_name = name;
66 int counter = 1;
67 while (std::filesystem::exists(dir / (unique_name + ".json"))) {
68 unique_name = name + "_" + std::to_string(counter++);
69 }
70
71 if (m_browser_system.create_material(dir, unique_name)) {
72 refresh();
73 }
74 }

80 std::filesystem::path new_path =

107 void asset_browser_view_model::request_move(
108 const std::filesystem::path& src, const std::filesystem::path& dst_dir)
109 {
110 try {
111 std::filesystem::path dst = dst_dir / src.filename();
112 if (src != dst && std::filesystem::exists(src)) {
113 std::filesystem::rename(src, dst);
114 refresh();
115 }
116 }
117 catch (const std::exception& e) {
118 // Sessizce yutulan hata düzeltildi. Dosya kilidi (lock) veya izin sorunlarında uyarır.
119 std::cout << "[Vesta AssetBrowser] Hata: Dosya tasinirken bir sorun olustu: " << e.what() << "\n";
120 }
121 }
