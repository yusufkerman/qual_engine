67void render_system::render(
68 const registry& reg,
69 const light_batch& lightBatch,
70 EngineContext& ctx,
71 const glm::mat4& view,
72 const glm::mat4& projection)
73{
74 // Mevcut durumları (FBO ve Viewport) işlemler bitince geri dönmek üzere yedekle
75 GLint previousFBO = 0;
76 glGetIntegerv(GL_FRAMEBUFFER_BINDING, &previousFBO);
77
78 GLint previousViewport[4];
79 glGetIntegerv(GL_VIEWPORT, previousViewport);
80
81 glBindBufferBase(GL_UNIFORM_BUFFER, 0, m_LightUBO);
82
83 const glm::mat4 viewProj = projection * view;
84 const glm::mat4 invView = glm::inverse(view);
85 const glm::vec3 cameraPosition = glm::vec3(invView[3]);
86
87 // 1. Işık Verilerinin GPU'ya Aktarımı (UBO)
88 const auto& lights = lightBatch.lights;
89 const int lightCount = std::min(static_cast<int>(lights.size()), static_cast<int>(MAX_LIGHT_COUNT));
90
91 glBindBuffer(GL_UNIFORM_BUFFER, m_LightUBO);
92 glBufferSubData(GL_UNIFORM_BUFFER, 0, sizeof(int), &lightCount);
93
94 if (lightCount > 0)
95 {
96 glBufferSubData(GL_UNIFORM_BUFFER, 16, lightCount * sizeof(light_data), lights.data());
97 }
98 glBindBuffer(GL_UNIFORM_BUFFER, 0);
99
100 // 2. Çıkarım (Extraction) ve Gruplama (Batching)
101 m_RenderItemsCache.clear();
102 extract(reg, ctx, viewProj);
103
104 m_RenderQueueCache.batches.clear();
105
106 // Varlıklar, Aria'dan çözümlenen GPU Shader ID'sine göre gruplanır (State değişimi minimizasyonu)
107 for (const auto& item : m_RenderItemsCache)
108 {
109 m_RenderQueueCache.batches[item.gpu_shader_id].push_back(item);
110 }
111
112 // 3. Gölge Haritası için Işık Matrisinin Hesaplanması (Directional Light araması)
113 glm::mat4 lightSpaceMatrix(1.0f);
114 bool hasDirectionalLight = false;
115
116 for (int i = 0; i < lightCount; ++i)
117 {
118 if (lights[i].type == 0) // Directional
119 {
120 lightSpaceMatrix = calculate_light_space_matrix(lights[i], cameraPosition);
121 hasDirectionalLight = true;
122 break;
123 }
124 }
125
126 // ========================================================================
127 // GÖLGE GEÇİŞİ (SHADOW PASS)
128 // ========================================================================
129 if (hasDirectionalLight)
130 {
131 ctx.shadowFBO.bind();
132
133 const unsigned int shadowGLShaderID = ctx.aria_system->get_shader_id(ctx.shadowShaderHandle);
134 glUseProgram(shadowGLShaderID);
135
136 glUniformMatrix4fv(glGetUniformLocation(shadowGLShaderID, "lightSpaceMatrix"), 1, GL_FALSE, &lightSpaceMatrix[0][0]);
137
138 for (auto& [gpuShaderID, batch] : m_RenderQueueCache.batches)
139 {
140 for (const auto& item : batch)
141 {
142 glUniformMatrix4fv(glGetUniformLocation(shadowGLShaderID, "model"), 1, GL_FALSE, &item.model[0][0]);
143 ctx.renderer.Draw(item, ctx.aria_system);
144 }
145 }
146
147 ctx.renderer.BindFramebuffer(previousFBO);
148 }
149
150 // ========================================================================
151 // ANA ÇİZİM GEÇİŞİ (MAIN RENDER PASS)
152 // ========================================================================
153 ctx.renderer.SetViewport(previousViewport[0], previousViewport[1], previousViewport[2], previousViewport[3]);
154
155 for (auto& [gpuShaderID, batch] : m_RenderQueueCache.batches)
156 {
157 glUseProgram(gpuShaderID);
158
159 // Küresel Kamera ve Gölge Üniformaları
160 glUniformMatrix4fv(glGetUniformLocation(gpuShaderID, "view"), 1, GL_FALSE, &view[0][0]);
161 glUniformMatrix4fv(glGetUniformLocation(gpuShaderID, "projection"), 1, GL_FALSE, &projection[0][0]);
162 glUniform3fv(glGetUniformLocation(gpuShaderID, "viewPos"), 1, &cameraPosition[0]);
163 glUniformMatrix4fv(glGetUniformLocation(gpuShaderID, "lightSpaceMatrix"), 1, GL_FALSE, &lightSpaceMatrix[0][0]);
164
165 // Sis (Fog) Parametreleri (İlerleyen süreçte EngineContext üzerinden dinamik alınabilir)
166 const bool fogEnabled = true;
167 const glm::vec3 fogColor = glm::vec3(0.7f, 0.8f, 0.9f);
168 const float fogDensity = 0.008f;
169
170 glUniform1i(glGetUniformLocation(gpuShaderID, "u_FogEnabled"), fogEnabled);
171 glUniform3fv(glGetUniformLocation(gpuShaderID, "u_FogColor"), 1, &fogColor[0]);
172 glUniform1f(glGetUniformLocation(gpuShaderID, "u_FogDensity"), fogDensity);
173
174 // Gölge haritası dokusunu 2. slota bağla
175 ctx.renderer.BindTextureSlot(2, ctx.shadowFBO.get_depth_texture());
176 glUniform1i(glGetUniformLocation(gpuShaderID, "shadowMap"), 2);
177
178 // Paket (Batch) içindeki varlıkların bireysel (Per-Object) çizimi
179 for (const auto& item : batch)
180 {
181 glUniformMatrix4fv(glGetUniformLocation(gpuShaderID, "model"), 1, GL_FALSE, &item.model[0][0]);
182 glUniform3fv(glGetUniformLocation(gpuShaderID, "baseColor"), 1, &item.base_color[0]);
183 glUniform2fv(glGetUniformLocation(gpuShaderID, "u_TextureScale"), 1, &item.texture_scale[0]);
184
185 glUniform1i(glGetUniformLocation(gpuShaderID, "useTexture"), item.diffuse_handle.is_valid());
186 glUniform1i(glGetUniformLocation(gpuShaderID, "useNormalMap"), item.normal_handle.is_valid());
187
188 if (item.normal_handle.is_valid())
189 {
190 glUniform1f(glGetUniformLocation(gpuShaderID, "normalStrength"), item.normal_strength);
191 }
192
193 glUniform1f(glGetUniformLocation(gpuShaderID, "specularStrength"), item.specular_strength);
194 glUniform1f(glGetUniformLocation(gpuShaderID, "shininess"), item.shininess);
195
196 glUniform1i(glGetUniformLocation(gpuShaderID, "u_RenderMode"), static_cast<int>(item.mode));
197 glUniform1f(glGetUniformLocation(gpuShaderID, "u_AlphaThreshold"), item.alpha_threshold);
198
199 ctx.renderer.Draw(item, ctx.aria_system);
200 }
201 }

204void render_system::extract(
205 const registry& reg,
206 EngineContext& ctx,
207 const glm::mat4& viewProj)
208{
209 const frustum f = buildFrustum(viewProj);
210 constexpr float kCullBias = 0.01f;
211
212 for (const auto& [e, meshComp] : reg.get_all<mesh_component>())
213 {
214 const auto* transform = reg.get_component<transform_component>(e);
215 const auto* matComp = reg.get_component<material_component>(e);
216
217 if (!transform)
218 {
219 continue;
220 }
221
222 // 1. Model Matrisinin Hesaplanması
223 const glm::mat4 model = ecs_transform::get_world_matrix(e, reg);
224
225 // 2. Frustum Culling (Görüş Alanı Kırpması)
226 const aria::submesh_runtime_info meshInfo =
227 ctx.aria_system->get_submesh_info(
228 meshComp.model_handle,
229 meshComp.sub_mesh_index);
230
231 const glm::vec3 localCenter = meshInfo.local_center;
232 const glm::vec4 homogeneousCenter = model * glm::vec4(localCenter, 1.0f);
233 const glm::vec3 worldCenter = glm::vec3(homogeneousCenter) / homogeneousCenter.w;
234
235 const glm::vec3 absoluteScale = glm::abs(transform->scale);
236 const float maxScale = glm::max(
237 absoluteScale.x,
238 glm::max(absoluteScale.y, absoluteScale.z));
239
240 const float radius = (meshInfo.local_radius * maxScale) + kCullBias;
241
242 if (!f.is_inside(worldCenter, radius))
243 {
244 continue;
245 }
246
247 // 3. Render Öğe Verilerinin Toplanması
248 render_item item;
249 item.model = model;
250 item.model_handle = meshComp.model_handle;
251 item.sub_mesh_index = meshComp.sub_mesh_index;
252
253 if (matComp && matComp->material_handle.is_valid())
254 {
255 const auto* mat = ctx.aria_system->get_material_data(matComp->material_handle);
256
257 if (mat)
258 {
259 if (mat->shader_handle.is_valid())
260 {
261 item.gpu_shader_id = ctx.aria_system->get_shader_id(mat->shader_handle);
262 }
263 else
264 {
265 item.gpu_shader_id = ctx.aria_system->get_shader_id(aria::handle<aria::shader_asset>());
266 }
267
268 item.mode = mat->mode;
269 item.alpha_threshold = mat->alpha_threshold;
270 item.base_color = mat->base_color;
271 item.normal_strength = mat->normal_strength;
272 item.texture_scale = mat->texture_scale;
273 item.diffuse_handle = mat->diffuse_texture_handle;
274 item.normal_handle = mat->normal_texture_handle;
275 item.specular_handle = mat->specular_texture_handle;
276 item.specular_strength = mat->specular_strength;
277 item.shininess = mat->shininess;
278 }
279 else
280 {
281 // Varlık çözümlemesi başarısız olursa güvenli varsayılan değerler atanır
282 item.gpu_shader_id = ctx.aria_system->get_shader_id(aria::handle<aria::shader_asset>());
283 item.mode = aria::render_mode::opaque;
284 item.alpha_threshold = 0.1f;
285 item.base_color = glm::vec3(1.0f, 0.0f, 1.0f);
286 item.texture_scale = glm::vec2(1.0f);
287 item.specular_strength = 0.5f;
288 item.shininess = 32.0f;
289 }
290 }
291 else
292 {
293 // Materyal bileşeni yoksa varsayılan değerler atanır
294 item.gpu_shader_id = ctx.aria_system->get_shader_id(aria::handle<aria::shader_asset>());
295 item.mode = aria::render_mode::opaque;
296 item.alpha_threshold = 0.1f;
297 item.base_color = glm::vec3(1.0f);
298 item.texture_scale = glm::vec2(1.0f);
299 item.specular_strength = 0.5f;
300 item.shininess = 32.0f;
301 }
302
303 m_RenderItemsCache.push_back(item);
304 }
305}

307glm::mat4 render_system::calculate_light_space_matrix(const light_data& light, const glm::vec3& viewPos)
308{
309 const float near_plane = 0.1f;
310 const float far_plane = 500.0f;
311 const glm::mat4 lightProjection = glm::ortho(-150.0f, 150.0f, -150.0f, 150.0f, near_plane, far_plane);
312
313 glm::vec3 lightPos = light.position;
314
315 if (light.type == 0) // Yönlü Işık (Directional Light)
316 {
317 // Kameranın dikey hareketlerinden etkilenmemesi ve gölge sınırlarının
318 // kamerayı dengeli takip etmesi için ışık merkezi kameranın yatay düzlemine (X, Z) sabitlenir.
319 const glm::vec3 shadowCenter = glm::vec3(viewPos.x, 0.0f, viewPos.z);
320
321 // Sabit yükseklik (100.0f) belirlenip, ışık yönünün tersine doğru izdüşüm mesafesi kadar geri çekilir.
322 lightPos = shadowCenter + glm::vec3(0.0f, 100.0f, 0.0f) - (glm::normalize(light.direction) * 150.0f);
323
324 const glm::vec3 target = shadowCenter + light.direction;
325 const glm::mat4 lightView = glm::lookAt(lightPos, target, glm::vec3(0.0f, 1.0f, 0.0f));
326
327 return lightProjection * lightView;
328 }
329
330 const glm::vec3 target = lightPos + light.direction;
331 const glm::mat4 lightView = glm::lookAt(lightPos, target, glm::vec3(0.0f, 1.0f, 0.0f));
332
333 return lightProjection * lightView;
334}
