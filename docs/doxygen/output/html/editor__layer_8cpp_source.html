41 void update_viewports(EngineContext& ctx)
42 {
43 auto& viewportState = ctx.vestaSys.viewport_state();
44
45 // 1. Sahne (Scene) Hedef Çözünürlüğü
46 const auto& scene = viewportState.scene();
47 if (scene.size.x > 1.0f && scene.size.y > 1.0f)
48 {
49 ctx.editorTarget.resize(
50 static_cast<int>(scene.size.x),
51 static_cast<int>(scene.size.y)
52 );
53 }
54
55 // 2. Oyun (Game) Hedef Çözünürlüğü
56 const auto& game = viewportState.game();
57 if (game.size.x > 1.0f && game.size.y > 1.0f)
58 {
59 auto& gameState = viewportState.game();
60 constexpr float renderScale = 1.0f;
61
62 gameState.render_size.x = std::max(1, static_cast<int>(gameState.size.x * renderScale));
63 gameState.render_size.y = std::max(1, static_cast<int>(gameState.size.y * renderScale));
64
65 ctx.gameTarget.resize(
66 gameState.render_size.x,
67 gameState.render_size.y
68 );
69 }
70 }
