10void MeshRenderer::Draw(const render_item& item, aria::system* ariaSystem)
11{
12 if (!ariaSystem) return;
13
14 // 1. Ağ (Mesh) indis sayısını al. Çizilecek veri yoksa erken çıkış yap.
15 const unsigned int indexCount = ariaSystem->get_mesh_index_count(item.model_handle, item.sub_mesh_index);
16 if (indexCount == 0) return;
17
18 // 2. Aria önbelleğinden GPU VAO kimliğini çek.
19 const aria::submesh_runtime_info meshInfo = ariaSystem->get_submesh_info(item.model_handle, item.sub_mesh_index);
20 const unsigned int targetVAO = meshInfo.gpu_id.vao_or_buffer_id;
21
22 if (targetVAO == 0) return;
23
24 // 3. VAO DURUM ÖNBELLEĞİ (STATE CACHING)
25 if (m_currentVAO != targetVAO)
26 {
27 glBindVertexArray(targetVAO);
28 m_currentVAO = targetVAO;
29 }
30
31 // 4. DOKU DURUM ÖNBELLEĞİ
32 auto bindTextureIfValid = [&](const auto& handle, unsigned int slot)
33 {
34 if (handle.is_valid())
35 {
36 const unsigned int texID = ariaSystem->get_texture_id(handle);
37 if (m_currentTextures[slot] != texID)
38 {
39 glActiveTexture(GL_TEXTURE0 + slot);
40 glBindTexture(GL_TEXTURE_2D, texID);
41 m_currentTextures[slot] = texID;
42 }
43 }
44 };
45
46 bindTextureIfValid(item.diffuse_handle, RenderConstants::DIFFUSE_TEXTURE_SLOT);
47 bindTextureIfValid(item.normal_handle, RenderConstants::NORMAL_TEXTURE_SLOT);
48 bindTextureIfValid(item.specular_handle, RenderConstants::SPECULAR_TEXTURE_SLOT);
49
50 // 5. ÇİZİM ÇAĞRISI (Draw Call)
51 glDrawElements(GL_TRIANGLES, static_cast<GLsizei>(indexCount), GL_UNSIGNED_INT, reinterpret_cast<const void*>(0));
52
53 // ========================================================================
54 // 6. GÜVENLİK TEMİZLİĞİ (SAFE UNBIND)
55 // Diğer sistemlerin (Skybox, Gizmo, UI) güncel durumu yanlış yorumlamaması
56 // için çizim sonrası bağlı kaynakları serbest bırakıyoruz.
57 // ========================================================================
58 if (item.diffuse_handle.is_valid()) {
59 glActiveTexture(GL_TEXTURE0 + RenderConstants::DIFFUSE_TEXTURE_SLOT);
60 glBindTexture(GL_TEXTURE_2D, 0);
61 m_currentTextures[RenderConstants::DIFFUSE_TEXTURE_SLOT] = 0;
62 }
63 if (item.normal_handle.is_valid()) {
64 glActiveTexture(GL_TEXTURE0 + RenderConstants::NORMAL_TEXTURE_SLOT);
65 glBindTexture(GL_TEXTURE_2D, 0);
66 m_currentTextures[RenderConstants::NORMAL_TEXTURE_SLOT] = 0;
67 }
68 if (item.specular_handle.is_valid()) {
69 glActiveTexture(GL_TEXTURE0 + RenderConstants::SPECULAR_TEXTURE_SLOT);
70 glBindTexture(GL_TEXTURE_2D, 0);
71 m_currentTextures[RenderConstants::SPECULAR_TEXTURE_SLOT] = 0;
72 }
73
74 glBindVertexArray(0);
75 m_currentVAO = 0;
76}
