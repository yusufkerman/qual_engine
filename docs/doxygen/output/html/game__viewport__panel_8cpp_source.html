19 void game_viewport_panel::render()
20 {
21 ImGui::Begin(m_title.c_str(), &m_is_open);
22
23 // Çekirdek editör durumundan oyun ekranı verisine ulaş
24 auto& state = m_viewport_state.game();
25
26 // Girdi (Input) ve kısayolların doğru çalışması için odak/imleç durumlarını güncelle
27 state.focused = ImGui::IsWindowFocused(ImGuiFocusedFlags_RootAndChildWindows);
28 state.hovered = ImGui::IsWindowHovered(ImGuiHoveredFlags_RootAndChildWindows);
29
30 ImGui::PushID("GameViewport");
31
32 // Render hedefinin pencere kenar boşlukları olmadan alana tam sığması için padding'i sıfırla
33 ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(0, 0));
34 ImGui::PushStyleVar(ImGuiStyleVar_WindowBorderSize, 0.0f);
35
36 ImVec2 avail = ImGui::GetContentRegionAvail();
37
38 if (avail.x > 1.0f && avail.y > 1.0f)
39 {
40 state.size = { avail.x, avail.y };
41 }
42
43 unsigned int tex = m_vm.get_texture_id();
44
45 if (tex != 0 && avail.x > 0.0f && avail.y > 0.0f)
46 {
47 ImVec2 screenPos = ImGui::GetCursorScreenPos();
48 state.position = { screenPos.x, screenPos.y };
49
50 // OpenGL dokularının Y ekseni ters olduğu için UV koordinatlarını (0,1 -> 1,0) ters çevirerek çiz
51 ImGui::Image(
52 (ImTextureID)(intptr_t)tex,
53 avail,
54 ImVec2(0, 1),
55 ImVec2(1, 0)
56 );
57 }
58
59 ImGui::PopStyleVar(2);
60 ImGui::PopID();
61 ImGui::End();
62 }
