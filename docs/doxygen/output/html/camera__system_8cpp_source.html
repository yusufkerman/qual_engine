11void camera_system::update(scene& scn)
12{
13 auto& reg = scn.get_registry();
14
15 entity bestCamera = 0;
16 bool foundPrimary = false;
17
18 // 1. Birincil kamerayı bul ve birden fazla primary varsa fazlalıkları sıfırla
19 for (const entity e : scn.get_entities())
20 {
21 if (!reg.has_component<camera_component>(e))
22 {
23 continue;
24 }
25
26 auto* cam = reg.get_component<camera_component>(e);
27
28 if (cam->primary)
29 {
30 if (!foundPrimary)
31 {
32 foundPrimary = true;
33 bestCamera = e;
34 }
35 else
36 {
37 cam->primary = false;
38 }
39 }
40 }
41
42 // 2. Birincil kamera atanmamışsa bulunan ilk kamerayı birincil yap
43 if (!foundPrimary)
44 {
45 for (const entity e : scn.get_entities())
46 {
47 if (!reg.has_component<camera_component>(e))
48 {
49 continue;
50 }
51
52 reg.get_component<camera_component>(e)->primary = true;
53 bestCamera = e;
54 break;
55 }
56 }
57
58 // 3. Değişiklik tespiti ve aktif kameranın atanması
59 const entity prev = m_lastCamera;
60
61 if (bestCamera != 0)
62 {
63 scn.set_active_camera(bestCamera);
64 }
65
66 if (bestCamera != prev)
67 {
68 std::cout << "[CameraSystem] Aktif kamera degisti: "
69 << prev << " -> " << bestCamera << '\n';
70 m_lastCamera = bestCamera;
71 }
72}
