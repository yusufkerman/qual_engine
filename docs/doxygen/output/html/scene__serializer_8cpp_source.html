20bool scene_serializer::serialize(const std::string& filepath, const scene& currentScene, EngineContext& ctx)
21{
22 const registry& reg = currentScene.get_registry();
23 const auto& allEntities = reg.get_all_entities();
24
25 json sceneJson;
26 sceneJson["scene_name"] = "Aria_Engine_Scene";
27 sceneJson["active_camera"] = currentScene.get_active_camera();
28
29 // 1. Doku referanslarının serileştirilmesi (Legacy doku kayıt defteri)
30 sceneJson["textures"] = json::array();
31 const auto& allTexturePaths = ctx.textureManager.get_all_paths();
32 for (const auto& [texId, path] : allTexturePaths)
33 {
34 json texJson;
35 texJson["id"] = texId;
36 texJson["path"] = path;
37 sceneJson["textures"].push_back(texJson);
38 }
39
40 // 2. Varlık ve bileşen verilerinin serileştirilmesi
41 sceneJson["entities"] = json::array();
42 for (const entity e : allEntities)
43 {
44 json entityJson;
45 entityJson["id"] = e;
46
47 serialize_component_if_exists<name_component>(e, reg, entityJson, "name_component");
48 serialize_component_if_exists<transform_component>(e, reg, entityJson, "transform_component");
49 serialize_component_if_exists<mesh_component>(e, reg, entityJson, "mesh_component");
50 serialize_component_if_exists<material_component>(e, reg, entityJson, "material_component");
51 serialize_component_if_exists<camera_component>(e, reg, entityJson, "camera_component");
52 serialize_component_if_exists<light_component>(e, reg, entityJson, "light_component");
53 serialize_component_if_exists<bounds_component>(e, reg, entityJson, "bounds_component");
54 serialize_component_if_exists<rigidbody_component>(e, reg, entityJson, "rigidbody_component");
55 serialize_component_if_exists<collider_component>(e, reg, entityJson, "collider_component");
56 serialize_component_if_exists<native_script_component>(e, reg, entityJson, "native_script_component");
57
58 sceneJson["entities"].push_back(entityJson);
59 }
60
61 std::ofstream file(filepath);
62 if (!file.is_open())
63 {
64 std::cerr << "[Serializer] Dosya acilamadi: " << filepath << '\n';
65 return false;
66 }
67
68 file << sceneJson.dump(4);
69 file.close();
70
71 std::cout << "[Serializer] Sahne basariyla kaydedildi: " << filepath << '\n';
72 return true;
73}

75bool scene_serializer::deserialize(const std::string& filepath, scene& currentScene, EngineContext& ctx)
76{
77 std::ifstream file(filepath);
78 if (!file.is_open())
79 {
80 std::cerr << "[Serializer] Dosya bulunamadi: " << filepath << '\n';
81 return false;
82 }
83
84 json sceneJson;
85 try
86 {
87 file >> sceneJson;
88 }
89 catch (const json::parse_error& e)
90 {
91 std::cerr << "[Serializer] JSON parse hatasi: " << e.what() << '\n';
92 file.close();
93 return false;
94 }
95 file.close();
96
97 // Sahneyi ve sistem önbelleklerini sıfırla
98 currentScene.new_scene();
99
100 if (ctx.aria_system)
101 {
102 ctx.aria_system->clear_all_resources();
103 }
104
105 ctx.textureManager.clear();
106 ctx.materialManager.clear();
107 ctx.assetManager.clear();
108
109 ctx.physicsSystem.shutdown();
110 ctx.physicsSystem.init(ctx.session.editor_scene());
111
112 std::unordered_map<unsigned int, entity> entityIdMap;
113 registry& reg = currentScene.get_registry();
114
115 // 1. Aşama: Varlık tanıtıcılarını oluştur
116 if (sceneJson.contains("entities"))
117 {
118 for (const auto& entityJson : sceneJson["entities"])
119 {
120 const entity newE = currentScene.create_entity();
121 entityIdMap[entityJson["id"].get<unsigned int>()] = newE;
122 }
123
124 // 2. Aşama: Bileşenleri yeni varlıklara bağla
125 for (const auto& entityJson : sceneJson["entities"])
126 {
127 const entity newE = entityIdMap[entityJson["id"].get<unsigned int>()];
128
129 deserialize_component_if_exists<name_component>(newE, reg, entityJson, "name_component");
130 deserialize_component_if_exists<transform_component>(newE, reg, entityJson, "transform_component");
131 deserialize_component_if_exists<camera_component>(newE, reg, entityJson, "camera_component");
132 deserialize_component_if_exists<light_component>(newE, reg, entityJson, "light_component");
133 deserialize_component_if_exists<bounds_component>(newE, reg, entityJson, "bounds_component");
134 deserialize_component_if_exists<rigidbody_component>(newE, reg, entityJson, "rigidbody_component");
135 deserialize_component_if_exists<collider_component>(newE, reg, entityJson, "collider_component");
136
137 if (entityJson.contains("native_script_component"))
138 {
139 const std::string s_name = entityJson["native_script_component"].value("script_name", "None");
140 if (s_name != "None")
141 {
142 script_factory::instantiate_script(s_name, newE, reg);
143 }
144 }
145
146 if (entityJson.contains("mesh_component"))
147 {
148 mesh_component meshComp = entityJson["mesh_component"].get<mesh_component>();
149 reg.add_component<mesh_component>(newE, meshComp);
150 }
151
152 if (entityJson.contains("material_component"))
153 {
154 material_component matComp = entityJson["material_component"].get<material_component>();
155 reg.add_component<material_component>(newE, matComp);
156 }
157 }
158 }
159
160 // Aktif kamera ve ışık durumunu senkronize et
161 if (sceneJson.contains("active_camera"))
162 {
163 const unsigned int oldCamId = sceneJson["active_camera"].get<unsigned int>();
164 const auto it = entityIdMap.find(oldCamId);
165 if (it != entityIdMap.end())
166 {
167 currentScene.set_active_camera(it->second);
168 }
169 }
170
171 currentScene.update_light_batch();
172 return true;
173}
