16 gpu_texture_id opengl_backend::create_texture(const raw_texture_data& raw_data) {
17 GLenum format = GL_RGB;
18 if (raw_data.channels == 1) format = GL_RED;
19 else if (raw_data.channels == 3) format = GL_RGB;
20 else if (raw_data.channels == 4) format = GL_RGBA;
21
22 GLuint tex_id;
23 glGenTextures(1, &tex_id);
24 glBindTexture(GL_TEXTURE_2D, tex_id);
25
26 glTexImage2D(GL_TEXTURE_2D, 0, format, raw_data.width, raw_data.height, 0, format, GL_UNSIGNED_BYTE, raw_data.pixels.data());
27 glGenerateMipmap(GL_TEXTURE_2D);
28
29 // Doku Filtreleme Ayarları
30 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
31 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
32 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
33 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
34
35 // Anizotropik Filtreleme
36 float max_aniso = 0.0f;
37 glGetFloatv(GL_MAX_TEXTURE_MAX_ANISOTROPY, &max_aniso);
38 glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_MAX_ANISOTROPY, glm::min(8.0f, max_aniso));
39
40 glBindTexture(GL_TEXTURE_2D, 0);
41 return tex_id;
42 }

44 std::vector<submesh_bridge> opengl_backend::create_model(const raw_model_data& raw_data) {
45 std::vector<submesh_bridge> out_submeshes;
46 out_submeshes.reserve(raw_data.sub_meshes.size());
47
48 for (const auto& raw_mesh : raw_data.sub_meshes) {
49 submesh_bridge submesh;
50
51 glGenVertexArrays(1, &submesh.gpu_data.vao_or_buffer_id);
52 glGenBuffers(1, &submesh.gpu_data.vbo);
53 glGenBuffers(1, &submesh.gpu_data.ebo);
54
55 glBindVertexArray(submesh.gpu_data.vao_or_buffer_id);
56
57 glBindBuffer(GL_ARRAY_BUFFER, submesh.gpu_data.vbo);
58 glBufferData(GL_ARRAY_BUFFER, raw_mesh.vertices.size() * sizeof(raw_vertex), raw_mesh.vertices.data(), GL_STATIC_DRAW);
59
60 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, submesh.gpu_data.ebo);
61 glBufferData(GL_ELEMENT_ARRAY_BUFFER, raw_mesh.indices.size() * sizeof(uint32_t), raw_mesh.indices.data(), GL_STATIC_DRAW);
62
63 // Pozisyon (location = 0)
64 glEnableVertexAttribArray(0);
65 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, sizeof(raw_vertex), (void*)offsetof(raw_vertex, position));
66
67 // Normal (location = 1)
68 glEnableVertexAttribArray(1);
69 glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, sizeof(raw_vertex), (void*)offsetof(raw_vertex, normal));
70
71 // UV Koordinatları (location = 2)
72 glEnableVertexAttribArray(2);
73 glVertexAttribPointer(2, 2, GL_FLOAT, GL_FALSE, sizeof(raw_vertex), (void*)offsetof(raw_vertex, tex_coord));
74
75 // Tanjant (location = 3)
76 glEnableVertexAttribArray(3);
77 glVertexAttribPointer(3, 3, GL_FLOAT, GL_FALSE, sizeof(raw_vertex), (void*)offsetof(raw_vertex, tangent));
78
79 submesh.bounds.local_center = raw_mesh.local_center;
80 submesh.bounds.local_radius = raw_mesh.local_radius;
81
82 glBindVertexArray(0);
83 out_submeshes.push_back(submesh);
84 }
85
86 return out_submeshes;
87 }

116 gpu_shader_id opengl_backend::create_shader(const raw_shader_data& raw_data) {
117 if (raw_data.vertex_source.empty() || raw_data.fragment_source.empty()) {
118 std::cout << "[ARIA OpenGL] Hata: Shader kaynak kodlari bos!\n";
119 return 0;
120 }
121
122 const char* vShaderCode = raw_data.vertex_source.c_str();
123 const char* fShaderCode = raw_data.fragment_source.c_str();
124
125 int success;
126 char infoLog[512];
127
128 // Vertex Shader Derleme
129 unsigned int vs = glCreateShader(GL_VERTEX_SHADER);
130 glShaderSource(vs, 1, &vShaderCode, NULL);
131 glCompileShader(vs);
132 glGetShaderiv(vs, GL_COMPILE_STATUS, &success);
133 if (!success) {
134 glGetShaderInfoLog(vs, 512, NULL, infoLog);
135 std::cout << "[ARIA OpenGL] Vertex Shader Derleme Hatasi:\n" << infoLog << "\n";
136 glDeleteShader(vs);
137 return 0;
138 }
139
140 // Fragment Shader Derleme
141 unsigned int fs = glCreateShader(GL_FRAGMENT_SHADER);
142 glShaderSource(fs, 1, &fShaderCode, NULL);
143 glCompileShader(fs);
144 glGetShaderiv(fs, GL_COMPILE_STATUS, &success);
145 if (!success) {
146 glGetShaderInfoLog(fs, 512, NULL, infoLog);
147 std::cout << "[ARIA OpenGL] Fragment Shader Derleme Hatasi:\n" << infoLog << "\n";
148 glDeleteShader(vs);
149 glDeleteShader(fs);
150 return 0;
151 }
152
153 // Program Baglama (Linking)
154 GLuint programID = glCreateProgram();
155 glAttachShader(programID, vs);
156 glAttachShader(programID, fs);
157 glLinkProgram(programID);
158
159 glGetProgramiv(programID, GL_LINK_STATUS, &success);
160 if (!success) {
161 glGetProgramInfoLog(programID, 512, NULL, infoLog);
162 std::cout << "[ARIA OpenGL] Program Baglama Hatasi:\n" << infoLog << "\n";
163 glDeleteShader(vs);
164 glDeleteShader(fs);
165 glDeleteProgram(programID);
166 return 0;
167 }
168
169 // UBO (Uniform Buffer Object) Entegrasyonu
170 unsigned int blockIndex = glGetUniformBlockIndex(programID, "LightBuffer");
171 if (blockIndex != GL_INVALID_INDEX) {
172 glUniformBlockBinding(programID, blockIndex, 0);
173 }
174 else {
175 std::cout << "[ARIA OpenGL] Uyari: LightBuffer uniform blogu shader (" << programID << ") icinde bulunamadi!\n";
176 }
177
178 glDeleteShader(vs);
179 glDeleteShader(fs);
180
181 return programID;
182 }

231 void opengl_backend::reset_state() {
232 // OpenGL durum makinesini (state machine) varsayılana döndür
233 glBindVertexArray(0);
234 glBindBuffer(GL_ARRAY_BUFFER, 0);
235 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);

241 glBindTexture(GL_TEXTURE_2D, 0);
242 }
