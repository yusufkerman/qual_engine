59 inline void apply_world_matrix_as_local(transform_component& t, const glm::mat4& world)
60 {
61 glm::vec3 skew;
62 glm::vec4 perspective;
63 glm::quat rotation_quat;
64 glm::vec3 translation, scale;
65
66 if (!glm::decompose(world, scale, rotation_quat, translation, skew, perspective))
67 {
68 return;
69 }
70
71 t.position = translation;
72 t.scale = scale;
73
74 const glm::mat3 R = glm::mat3_cast(rotation_quat);
75
76 float pitch, yaw, roll;
77
78 // GLM sütun-öncelikli (column-major) veri erişimi: R[col][row]
79 const float sinYaw = glm::clamp(R[2][0], -1.0f, 1.0f); // Matematiksel R[0][2]
80
81 if (sinYaw < 1.0f)
82 {
83 if (sinYaw > -1.0f)
84 {
85 pitch = std::atan2(-R[2][1], R[2][2]); // Math: -R[1][2], R[2][2]
86 yaw = std::asin(sinYaw);
87 roll = std::atan2(-R[1][0], R[0][0]); // Math: -R[0][1], R[0][0]
88 }
89 else
90 {
91 // Gimbal lock (Aşağı): sinYaw == -1
92 pitch = 0.0f;
93 yaw = -glm::half_pi<float>();
94 roll = -std::atan2(R[0][1], R[1][1]); // Math: R[1][0], R[1][1]
95 }
96 }
97 else
98 {
99 // Gimbal lock (Yukarı): sinYaw == 1
100 pitch = 0.0f;
101 yaw = glm::half_pi<float>();
102 roll = std::atan2(R[0][1], R[1][1]);
103 }
104
105 t.rotation = glm::vec3(pitch, yaw, roll);
106 }
