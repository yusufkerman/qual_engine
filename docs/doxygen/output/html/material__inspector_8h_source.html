40 inline void mat_parse_float_array(const std::string& content, const std::string& key, float* out, int count) {
41 size_t pos = content.find("\"" + key + "\"");
42 if (pos == std::string::npos) return;
43
44 size_t sb = content.find("[", pos);
45 if (sb == std::string::npos) return;
46
47 size_t eb = content.find("]", sb);
48 if (eb == std::string::npos || eb <= sb) return;
49
50 std::stringstream ss(content.substr(sb + 1, eb - sb - 1));
51 std::string tok;
52 int i = 0;
53 while (std::getline(ss, tok, ',') && i < count) {
54 try {
55 out[i] = std::stof(tok);
56 }
57 catch (...) {}
58 i++;
59 }
60 }

65 inline float mat_parse_float(const std::string& content, const std::string& key, float def) {
66 size_t pos = content.find("\"" + key + "\"");
67 if (pos == std::string::npos) return def;
68
69 size_t colon = content.find(":", pos);
70 if (colon == std::string::npos) return def;
71
72 size_t end_pos = content.find_first_of(",}\n\r", colon + 1);
73 if (end_pos == std::string::npos) end_pos = content.size();
74
75 std::string val_str = content.substr(colon + 1, end_pos - (colon + 1));
76 try { return std::stof(val_str); }
77 catch (...) { return def; }
78 }

101 inline std::string mat_parse_string(const std::string& content, const std::string& key, const std::string& def) {
102 size_t pos = content.find("\"" + key + "\"");
103 if (pos == std::string::npos) return def;
104
105 size_t colon = content.find(":", pos);
106 if (colon == std::string::npos) return def;
107
108 size_t q1 = content.find("\"", colon);
109 if (q1 == std::string::npos) return def;
110
111 size_t q2 = content.find("\"", q1 + 1);
112 if (q2 == std::string::npos) return def;
113
114 return content.substr(q1 + 1, q2 - q1 - 1);
115 }

120 inline std::string mat_resolve_guid_path(const std::string& guid_str, aria::system* aria_sys) {
121 if (guid_str.empty() || guid_str == "00000000-0000-0000-0000-000000000000" || !aria_sys) return "";
122 auto parsed = uuids::uuid::from_string(guid_str);

124 return "";
