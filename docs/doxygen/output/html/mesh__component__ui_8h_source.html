120inline void register_mesh_ui_system(aria::system& ariaSys) {
121 size_t comp_hash = typeid(mesh_component).hash_code();
122
123 // 1. ECS Operasyon Kaydı
124 vesta::ComponentOperationInfo logic;
125 logic.add_fn = add_mesh_component;
126
127 logic.has_fn = [](entity e, registry& reg) -> bool {
128 return reg.has_component<mesh_component>(e);
129 };
130
131 logic.remove_fn = [](entity e, registry& reg) {
132 reg.remove_component<mesh_component>(e);
133 };
134
135 vesta::component_operation_registry::register_component(comp_hash, logic);
136
137 // 2. UI Çizim Kaydı (Aria sistemi referansı closure içine yakalanır)
138 vesta::ComponentUIInfo ui;
139 ui.display_name = "Mesh Renderer";
140 ui.render_fn = [&ariaSys](entity e, registry& reg) {
141 render_mesh_ui(e, reg, ariaSys);
142 };
143
144 vesta::component_ui_registry::register_component(comp_hash, ui);
145}
