18 aria::asset_guid material_service::create_new_material(const std::filesystem::path& directory_path, const std::string& name) {
19 // Motorun standardı olan .json uzantısı ile dosya yolu oluşturulur
20 std::filesystem::path full_path = directory_path / (name + ".json");
21
22 std::ofstream file(full_path);
23 if (!file.is_open()) {
24 std::cerr << "[Vesta MaterialService] Hata: Materyal dosyasi olusturulamadi (" << full_path << ")\n";
25 return {};
26 }
27
28 // Varsayılan shader için sistemden GUID sorgulanır
29 aria::asset_guid default_shader_guid = asset_pipeline::get_or_create_meta_guid("Shaders/default_shader");
30
31 // Standart materyal JSON şablonu dosyaya yazılır
32 file << "{\n"
33 << " \"type\": \"material\",\n"
34 << " \"name\": \"" << name << "\",\n"
35 << " \"base_color\": [1.0, 1.0, 1.0, 1.0],\n"
36 << " \"specular_strength\": 0.5,\n"
37 << " \"shininess\": 32,\n"
38 << " \"normal_strength\": 1.0,\n"
39 << " \"shader_guid\": \"" << uuids::to_string(default_shader_guid) << "\",\n"
40 << " \"diffuse_texture_guid\": \"00000000-0000-0000-0000-000000000000\",\n"
41 << " \"normal_texture_guid\": \"00000000-0000-0000-0000-000000000000\",\n"
42 << " \"specular_texture_guid\": \"00000000-0000-0000-0000-000000000000\"\n"
43 << "}";
44 file.close();
45
46 // Dünya adaptörü (IEditableWorld) üzerinden bu yeni dosyaya ait GUID sorgulanır ve döndürülür
47 aria::asset_guid new_guid = m_world.get_guid_from_path(full_path.string());
48
49 std::cout << "[Vesta MaterialService] Yeni materyal olusturuldu: " << full_path.string() << "\n";
50
51 return new_guid;
52 }

54 aria::asset_guid material_service::duplicate_material(const aria::asset_guid& source_guid, const std::filesystem::path& target_directory) {
55
56 // IEditableWorld üzerinden kaynak GUID'in gerçek disk yolu çözümlenir
57 std::string source_path_str = m_world.resolve_asset_path(source_guid);
58 if (source_path_str.empty()) {
59 return {};
60 }
61
62 std::filesystem::path source_path(source_path_str);
63
64 // İsim çakışmasını önlemek için "_Copy" takısı eklenir
65 std::filesystem::path target_path = target_directory / (source_path.stem().string() + "_Copy" + source_path.extension().string());
66
67 int count = 1;
68 while (std::filesystem::exists(target_path)) {
69 target_path = target_directory / (source_path.stem().string() + "_Copy_" + std::to_string(count++) + source_path.extension().string());
70 }
71
72 try {
73 std::filesystem::copy_file(source_path, target_path, std::filesystem::copy_options::overwrite_existing);
74 }
75 catch (const std::filesystem::filesystem_error& e) {
76 std::cerr << "[Vesta MaterialService] Materyal kopyalama hatasi: " << e.what() << "\n";
77 return {};
78 }
79
80 // Yeni kopyalanan dosyayı sisteme tanıtıp motorun atadığı GUID döndürülür
81 return m_world.get_guid_from_path(target_path.string());
82 }

84 bool material_service::save_material_changes(const aria::asset_guid& guid, const std::string& serialized_data) {
85 if (guid.is_nil()) {
86 return false;
87 }
88
89 std::string path_str = m_world.resolve_asset_path(guid);
90 if (path_str.empty()) {
91 return false;
92 }
93
94 std::ofstream file(path_str);
95 if (!file.is_open()) {
96 std::cerr << "[Vesta MaterialService] Hata: Materyal dosyasi kayit icin acilamadi (" << path_str << ")\n";
97 return false;
98 }
99
100 file << serialized_data;
101 file.close();
102
103 // Sahnenin (dünyanın) değiştirildiğini sisteme bildir (Kaydedilmemiş değişiklikler için)
104 m_world.mark_scene_dirty();
105 return true;
106 }
