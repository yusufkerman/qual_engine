21 static void draw_entity_node(hierarchy_view_model* vm, const hierarchy_node_data& node)
22 {
23 ImGuiTreeNodeFlags flags = ImGuiTreeNodeFlags_OpenOnArrow | ImGuiTreeNodeFlags_SpanAvailWidth;
24
25 if (!node.has_children) {
26 flags |= ImGuiTreeNodeFlags_Leaf;
27 }
28
29 if (node.is_selected) {
30 flags |= ImGuiTreeNodeFlags_Selected;
31 }
32
33 bool opened = false;
34
35 // ==========================================
36 // YENİDEN ADLANDIRMA (RENAME) MODU
37 // ==========================================
38 if (vm->is_renaming(node.id))
39 {
40 static char rename_buffer[256];
41 static entity editing_entity = 0;
42
43 if (editing_entity != node.id)
44 {
45 editing_entity = node.id;
46
47 // Güvenli ve cross-platform (MSVC/GCC/Clang) metin kopyalama
48 snprintf(rename_buffer, sizeof(rename_buffer), "%s", vm->get_rename_buffer().c_str());
49 ImGui::SetKeyboardFocusHere();
50 }
51
52 ImGui::SetNextItemWidth(-1.0f);
53
54 bool commit = ImGui::InputText(
55 "##Rename",
56 rename_buffer,
57 sizeof(rename_buffer),
58 ImGuiInputTextFlags_EnterReturnsTrue | ImGuiInputTextFlags_AutoSelectAll
59 );
60
61 if (commit)
62 {
63 vm->get_rename_buffer() = rename_buffer;
64 vm->commit_rename();
65 editing_entity = 0;
66 }
67
68 if (ImGui::IsItemActive() && ImGui::IsKeyPressed(ImGuiKey_Escape))
69 {
70 vm->cancel_rename();
71 editing_entity = 0;
72 }
73
74 if (ImGui::IsItemDeactivatedAfterEdit())
75 {
76 vm->get_rename_buffer() = rename_buffer;
77 vm->commit_rename();
78 editing_entity = 0;
79 }
80
81 return; // Düzenleme modundayken alt hiyerarşiyi çizmeye devam etme
82 }
83 else
84 {
85 opened = ImGui::TreeNodeEx(
86 reinterpret_cast<void*>(static_cast<uintptr_t>(node.id)),
87 flags,
88 "%s",
89 node.name.c_str());
90 }
91
92 // ==========================================
93 // SEÇİM (SELECTION)
94 // ==========================================
95 if (ImGui::IsItemClicked()) {
96 vm->request_select(node.id, ImGui::GetIO().KeyCtrl);
97 }
98
99 // ==========================================
100 // SÜRÜKLEME KAYNAĞI (DRAG SOURCE)
101 // ==========================================
102 if (ImGui::BeginDragDropSource()) {
103 ImGui::SetDragDropPayload("VESTA_ENTITY", &node.id, sizeof(entity));
104 ImGui::Text("%s", node.name.c_str());
105 ImGui::EndDragDropSource();
106 }
107
108 // ==========================================
109 // BIRAKMA HEDEFİ (DROP TARGET)
110 // ==========================================
111 if (ImGui::BeginDragDropTarget()) {
112 if (const ImGuiPayload* payload = ImGui::AcceptDragDropPayload("VESTA_ENTITY")) {
113 entity dragged = *static_cast<const entity*>(payload->Data);
114 if (dragged != node.id) {
115 vm->request_set_parent(dragged, node.id);
116 }
117 }
118 ImGui::EndDragDropTarget();
119 }
120
121 // ==========================================
122 // ÇİFT TIKLA YENİDEN ADLANDIR
123 // ==========================================
124 if (ImGui::IsItemHovered() && ImGui::IsMouseDoubleClicked(ImGuiMouseButton_Left)) {
125 vm->start_rename(node.id);
126 }
127
128 // ==========================================
129 // İÇERİK MENÜSÜ (CONTEXT MENU)
130 // ==========================================
131 if (ImGui::BeginPopupContextItem()) {
132 if (ImGui::MenuItem("Rename")) {
133 vm->start_rename(node.id);
134 }
135 ImGui::Separator();
136 if (ImGui::MenuItem("Copy")) {
137 vm->request_copy(node.id);
138 }
139 if (ImGui::MenuItem("Duplicate")) {
140 vm->request_duplicate(node.id);
141 }
142 ImGui::Separator();
143 if (ImGui::MenuItem("Delete")) {
144 vm->request_destroy_entity(node.id);
145 }
146 ImGui::EndPopup();
147 }
148
149 // ==========================================
150 // ALT DÜĞÜMLERİN ÇİZİMİ (CHILDREN)
151 // ==========================================
152 if (opened) {
153 for (entity child : node.children) {
154 draw_entity_node(vm, vm->get_node_data(child));
155 }
156 ImGui::TreePop();
157 }
158 }

160 void hierarchy_panel::render()
161 {
162 if (!m_is_open) return;
163
164 ImGui::Begin(m_title.c_str(), &m_is_open);
165
166 // ==========================================
167 // KISAYOLLAR (SHORTCUTS)
168 // ==========================================
169 if (ImGui::IsWindowFocused(ImGuiFocusedFlags_RootAndChildWindows))
170 {
171 bool ctrl = ImGui::IsKeyDown(ImGuiKey_LeftCtrl) || ImGui::IsKeyDown(ImGuiKey_RightCtrl);
172
173 if (ctrl) {
174 if (ImGui::IsKeyPressed(ImGuiKey_C)) m_vm->request_copy_selected();
175 if (ImGui::IsKeyPressed(ImGuiKey_D)) m_vm->request_duplicate_selected();
176 if (ImGui::IsKeyPressed(ImGuiKey_V)) m_vm->request_paste();
177 }
178
179 if (ImGui::IsKeyPressed(ImGuiKey_Delete)) {
180 m_vm->request_delete_selected();
181 }
182
183 if (ImGui::IsKeyPressed(ImGuiKey_F2)) {
184 entity selected = m_vm->get_selected_entity();
185 if (selected != 0) {
186 m_vm->start_rename(selected);
187 }
188 }
189
190 if (ImGui::IsKeyPressed(ImGuiKey_Escape)) {
191 if (m_vm->get_renaming_entity() != 0) {
192 m_vm->cancel_rename();
193 }
194 }
195 }
196
197 // ==========================================
198 // ÜST ARAÇ ÇUBUĞU (TOOLBAR)
199 // ==========================================
200 if (ImGui::Button("Create Object")) {
201 m_vm->request_create_entity();
202 }
203
204 ImGui::Separator();
205
206 // ==========================================
207 // BOŞ ALAN İÇERİK MENÜSÜ (EMPTY AREA CONTEXT)
208 // ==========================================
209 if (ImGui::BeginPopupContextWindow("Hierarchy_Context", ImGuiPopupFlags_MouseButtonRight | ImGuiPopupFlags_NoOpenOverItems))
210 {
211 if (ImGui::MenuItem("Paste", nullptr, false, m_vm->can_paste())) {
212 m_vm->request_paste();
213 }
214 ImGui::EndPopup();
215 }
216
217 // ==========================================
218 // AĞAÇ ÇİZİMİ (DRAW HIERARCHY)
219 // ==========================================
220 const auto root_nodes = m_vm->get_root_nodes();
221 for (const auto& node : root_nodes) {
222 draw_entity_node(m_vm.get(), node);
223 }
224
225 // ==========================================
226 // KÖKE BIRAKMA ALANI (DROP TO ROOT)
227 // ==========================================
228 ImGui::InvisibleButton("##HierarchyRootDropZone", ImGui::GetContentRegionAvail());
229
230 if (ImGui::BeginDragDropTarget()) {
231 if (const ImGuiPayload* payload = ImGui::AcceptDragDropPayload("VESTA_ENTITY")) {
232 entity dragged = *static_cast<const entity*>(payload->Data);
233 m_vm->request_set_parent(dragged, 0); // 0 (Kök/Root) ebeveyne ata
234 }
235 ImGui::EndDragDropTarget();
236 }
237
238 ImGui::End();
239 }
